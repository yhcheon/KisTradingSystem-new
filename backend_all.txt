// 파일명: auth/jwt/jwks.rs
// 설명: Supabase ES256 JWKS 공개키 조회 및 캐시
// ============================================================

use std::{
    collections::HashMap,
    time::{Duration, Instant},
};

use jsonwebtoken::{DecodingKey, jwk::JwkSet};
use tokio::sync::{Mutex, OnceCell};

const JWKS_CACHE_TTL: Duration = Duration::from_secs(60 * 60);

struct CachedJwks {
    jwks: JwkSet,
    fetched_at: Instant,
}

static JWKS_CACHE: OnceCell<Mutex<HashMap<String, CachedJwks>>> = OnceCell::const_new();

pub(super) async fn get_es256_decoding_key(
    http_client: &reqwest::Client,
    supabase_url: &str,
    kid: &str,
) -> Result<DecodingKey, String> {
    let project_url = supabase_url.trim_end_matches('/').to_string();

    let cache = JWKS_CACHE
        .get_or_init(|| async { Mutex::new(HashMap::new()) })
        .await;

    let mut cache = cache.lock().await;

    let needs_refresh = cache
        .get(&project_url)
        .map(|cached| {
            cached.fetched_at.elapsed() >= JWKS_CACHE_TTL
                || !cached
                    .jwks
                    .keys
                    .iter()
                    .any(|key| key.common.key_id.as_deref() == Some(kid))
        })
        .unwrap_or(true);

    if needs_refresh {
        let url = format!("{project_url}/auth/v1/.well-known/jwks.json");
        tracing::info!(target: "auth", "Supabase JWKS 공개키 조회 시작");
        let response = http_client
            .get(&url)
            .send()
            .await
            .map_err(|e| format!("Supabase JWKS 요청 실패: {e}"))?
            .error_for_status()
            .map_err(|e| format!("Supabase JWKS HTTP 오류: {e}"))?;

        let jwks = response
            .json::<JwkSet>()
            .await
            .map_err(|e| format!("Supabase JWKS 응답 파싱 실패: {e}"))?;

        if jwks.keys.is_empty() {
            return Err("Supabase JWKS에 공개키가 없습니다".to_string());
        }

        tracing::info!(
            target: "auth",
            key_count = jwks.keys.len(),
            "Supabase JWKS 공개키 캐시 갱신 완료"
        );
        cache.insert(
            project_url.clone(),
            CachedJwks {
                jwks,
                fetched_at: Instant::now(),
            },
        );
    }

    let cached = cache
        .get(&project_url)
        .ok_or_else(|| "Supabase JWKS 캐시를 찾을 수 없습니다".to_string())?;
    let jwk = cached
        .jwks
        .keys
        .iter()
        .find(|key| key.common.key_id.as_deref() == Some(kid))
        .ok_or_else(|| {
            tracing::warn!(
                target: "auth",
                token_kid = %kid,
                "갱신된 JWKS에서도 JWT kid와 일치하는 공개키를 찾지 못함"
            );
            format!("JWKS에서 kid={kid} 공개키를 찾을 수 없습니다")
        })?;

    DecodingKey::from_jwk(jwk).map_err(|e| format!("JWKS 공개키 변환 실패: {e}"))
}

// ============================================================
// 파일명: auth/jwt/verify.rs
// 설명: Supabase ES256 및 Legacy HS256 JWT 검증
// ============================================================

use jsonwebtoken::{Algorithm, DecodingKey, Validation, decode, decode_header};

use super::{claims::Claims, jwks::get_es256_decoding_key};

/// Supabase JWT 검증. 토큰 원문이나 서명값은 로그에 기록하지 않는다.
pub async fn verify_jwt(
    token: &str,
    jwt_secret: &str,
    supabase_url: &str,
    http_client: &reqwest::Client,
) -> Result<Claims, String> {
    let header = decode_header(token).map_err(|error| {
        tracing::warn!(target: "auth", error = %error, "JWT 헤더 파싱 실패");
        format!("JWT 헤더 파싱 실패: {error}")
    })?;

    let token_alg = header.alg;
    let token_kid = header.kid.as_deref().unwrap_or("없음");
    tracing::debug!(
        target: "auth",
        token_alg = ?token_alg,
        token_kid = %token_kid,
        "JWT 검증 시작"
    );

    let (decoding_key, algorithm) = match token_alg {
        Algorithm::ES256 => {
            let kid = header
                .kid
                .as_deref()
                .ok_or_else(|| "ES256 JWT에 kid 헤더가 없습니다".to_string())?;
            (
                get_es256_decoding_key(http_client, supabase_url, kid).await?,
                Algorithm::ES256,
            )
        }
        Algorithm::HS256 => (
            DecodingKey::from_secret(jwt_secret.as_bytes()),
            Algorithm::HS256,
        ),
        other => {
            tracing::warn!(target: "auth", algorithm = ?other, "허용하지 않는 JWT 알고리즘");
            return Err(format!("허용하지 않는 JWT 알고리즘: {other:?}"));
        }
    };

    let mut validation = Validation::new(algorithm);
    validation.validate_aud = false;

    let data = decode::<Claims>(token, &decoding_key, &validation).map_err(|error| {
        tracing::warn!(
            target: "auth",
            token_alg = ?token_alg,
            token_kid = %token_kid,
            error = %error,
            "JWT 서명 또는 클레임 검증 실패"
        );
        format!("JWT 검증 실패 (alg={token_alg:?}, kid={token_kid}): {error}")
    })?;

    let expected_issuer = format!("{}/auth/v1", supabase_url.trim_end_matches('/'));
    if data.claims.iss.as_deref() != Some(expected_issuer.as_str()) {
        tracing::warn!(target: "auth", "JWT issuer 불일치");
        return Err("JWT issuer가 현재 Supabase 프로젝트와 일치하지 않습니다".to_string());
    }

    tracing::debug!(target: "auth", token_alg = ?token_alg, "JWT 검증 성공");
    Ok(data.claims)
}

// Redis 없이 프로세스 메모리에 주간/야간 가격을 독립 보관합니다.
use crate::kis::ws::RealtimeTick;
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::sync::Arc;
use std::time::Instant;
use tokio::sync::Mutex;

#[derive(Debug, Serialize, Deserialize, Clone)]
pub struct CachedPrice {
    pub code: String,
    pub name: String,
    pub price: String,
    pub prev_close: String,
    pub open: String,
    pub high: String,
    pub low: String,
    pub change: String,
    pub change_rate: String,
    pub volume: String,
    pub cached_at: u64,
}

#[derive(Clone, Default)]
pub struct PriceCache {
    // 종목코드만 키로 사용하던 구조를 (종목코드, 세션)으로 분리합니다.
    inner: Arc<Mutex<HashMap<(String, String), (CachedPrice, Instant)>>>,
}

impl PriceCache {
    pub fn new() -> Self {
        Self::default()
    }

    pub async fn get(&self, code: &str, session: &str) -> Option<CachedPrice> {
        let map = self.inner.lock().await;
        let (price, _) = map.get(&(code.to_string(), session.to_string()))?;
        Some(price.clone())
    }

    pub async fn set(&self, price: &CachedPrice, session: &str) {
        let mut map = self.inner.lock().await;
        map.insert(
            (price.code.clone(), session.to_string()),
            (price.clone(), Instant::now()),
        );
    }

    pub async fn clear_session(&self, session: &str) {
        let mut map = self.inner.lock().await;
        map.retain(|key, _| key.1.as_str() != session);
    }

    pub async fn update_by_tick(&self, tick: &RealtimeTick) {
        let mut map = self.inner.lock().await;
        let key = (tick.code.clone(), tick.market_session.clone());
        if let Some((cached, _)) = map.get_mut(&key) {
            cached.price = tick.price.clone();
            cached.volume = tick.volume.clone();
            cached.change = tick.change.clone();
            cached.change_rate = tick.change_rate.clone();

            let price = tick.price.parse::<f64>().unwrap_or(0.0);
            let high = cached.high.parse::<f64>().unwrap_or(0.0);
            let low = cached.low.parse::<f64>().unwrap_or(0.0);
            if price > high {
                cached.high = tick.price.clone();
            }
            if low == 0.0 || price < low {
                cached.low = tick.price.clone();
            }
        }
    }
}

impl NicknameConfig {
    pub fn from_file(path: &Path) -> Result<Self, ChatError> {
        let contents = fs::read_to_string(path).map_err(|error| {
            ChatError::Configuration(format!(
                "nickname config {} could not be read: {error}",
                path.display()
            ))
        })?;
        Self::from_json(&contents)
    }

    pub(super) fn from_json(contents: &str) -> Result<Self, ChatError> {
        let raw: RawNicknameConfig = serde_json::from_str(contents).map_err(|error| {
            ChatError::Configuration(format!("nickname config is invalid JSON: {error}"))
        })?;

        if raw.words.is_empty() {
            return Err(ChatError::Configuration(
                "nickname config words must not be empty".to_string(),
            ));
        }
        let mut words = Vec::with_capacity(raw.words.len());
        let mut seen = HashSet::with_capacity(raw.words.len());
        for word in raw.words {
            let word = word.trim();
            if word.is_empty() {
                return Err(ChatError::Configuration(
                    "nickname config words must not contain empty entries".to_string(),
                ));
            }
            if word.chars().count() > MAX_NICKNAME_WORD_LENGTH {
                return Err(ChatError::Configuration(format!(
                    "nickname config words must be at most {MAX_NICKNAME_WORD_LENGTH} characters"
                )));
            }
            if word.chars().any(char::is_control) {
                return Err(ChatError::Configuration(
                    "nickname config words must not contain control characters".to_string(),
                ));
            }
            if !seen.insert(word.to_string()) {
                return Err(ChatError::Configuration(
                    "nickname config words must be unique after trimming".to_string(),
                ));
            }
            words.push(word.to_string());
        }

        Ok(Self { words })
    }
}

fn validate_social_token(token: &str) -> Result<(), ChatError> {
    let length = token.chars().count();
    if token.trim() != token
        || length == 0
        || length > MAX_MESSAGE_CHARACTERS
        || token.starts_with('/')
        || token
            .chars()
            .any(|character| character.is_whitespace() || character.is_control())
    {
        return Err(ChatError::Configuration(
            "social commands and aliases must be non-empty single tokens without whitespace, control characters or a leading slash"
                .to_string(),
        ));
    }
    Ok(())
}

fn validate_social_template(template: &str, requires_argument: bool) -> Result<(), ChatError> {
    if template.trim().is_empty()
        || template.chars().count() > MAX_MESSAGE_CHARACTERS
        || template.chars().any(char::is_control)
    {
        return Err(ChatError::Configuration(
            "social templates must be non-empty, control-character-free and within the message length limit"
                .to_string(),
        ));
    }

    let mut has_nickname = false;
    let mut has_argument = false;
    let mut chars = template.chars().peekable();
    while let Some(character) = chars.next() {
        match character {
            '{' => {
                let mut placeholder = String::new();
                let mut closed = false;
                for next in chars.by_ref() {
                    if next == '}' {
                        closed = true;
                        break;
                    }
                    placeholder.push(next);
                }
                if !closed || !matches!(placeholder.as_str(), "nickname" | "argument" | "target") {
                    return Err(ChatError::Configuration(
                        "social templates may use only {nickname}, {target} and {argument} placeholders"
                            .to_string(),
                    ));
                }
                match placeholder.as_str() {
                    "nickname" => has_nickname = true,
                    "argument" => has_argument = true,
                    "target" => {}
                    _ => unreachable!(),
                }
            }
            '}' => {
                return Err(ChatError::Configuration(
                    "social templates contain an unmatched closing brace".to_string(),
                ));
            }
            _ => {}
        }
    }

    if !has_nickname || has_argument != requires_argument {
        return Err(ChatError::Configuration(if requires_argument {
            "social templates must contain both {nickname} and {argument}".to_string()
        } else {
            "social no-argument templates must contain {nickname} and must not contain {argument}"
                .to_string()
        }));
    }
    Ok(())
}

impl ChatCommandConfig {
    pub fn from_file(path: &Path) -> Result<Self, ChatError> {
        let contents = fs::read_to_string(path).map_err(|error| {
            ChatError::Configuration(format!(
                "chat command config {} could not be read: {error}",
                path.display()
            ))
        })?;
        Self::from_json(&contents)
    }

    pub(super) fn from_json(contents: &str) -> Result<Self, ChatError> {
        let raw: RawChatCommandConfig = serde_json::from_str(contents).map_err(|error| {
            ChatError::Configuration(format!("chat command config is invalid JSON: {error}"))
        })?;

        let actions = match raw.version {
            1 => {
                if !raw.actions.is_empty() {
                    return Err(ChatError::Configuration(
                        "version 1 chat command config must contain only commands".to_string(),
                    ));
                }
                HashMap::new()
            }
            2 => {
                if raw.actions.is_empty() {
                    return Err(ChatError::Configuration(
                        "version 2 chat command config actions must not be empty".to_string(),
                    ));
                }
                let mut registered_tokens = HashSet::new();
                let mut actions = HashMap::with_capacity(raw.actions.len());
                for (action_id, action) in raw.actions {
                    if action_id.is_empty()
                        || action_id.len() > 64
                        || !action_id.chars().all(|character| {
                            character.is_ascii_alphanumeric() || matches!(character, '_' | '-')
                        })
                    {
                        return Err(ChatError::Configuration(
                            "social action ids must use 1-64 ASCII letters, digits, '_' or '-'"
                                .to_string(),
                        ));
                    }

                    validate_social_token(&action.command)?;
                    if !registered_tokens.insert(action.command.clone()) {
                        return Err(ChatError::Configuration(format!(
                            "social command or alias is duplicated: {}",
                            action.command
                        )));
                    }
                    for alias in &action.aliases {
                        validate_social_token(alias)?;
                        if !registered_tokens.insert(alias.clone()) {
                            return Err(ChatError::Configuration(format!(
                                "social command or alias is duplicated: {alias}"
                            )));
                        }
                    }
                    validate_social_template(&action.template, true)?;
                    validate_social_template(&action.template_no_arg, false)?;
                    match (&action.template_target, &action.template_target_no_arg) {
                        (Some(with_argument), Some(without_argument)) => {
                            validate_social_template(with_argument, true)?;
                            validate_social_template(without_argument, false)?;
                            if !with_argument.contains("{target}") || !without_argument.contains("{target}") {
                                return Err(ChatError::Configuration(
                                    "target Social templates must contain {target}".to_string(),
                                ));
                            }
                            let expanded = without_argument
                                .replace("{nickname}", &"n".repeat(MAX_NICKNAME_ALIAS_LENGTH))
                                .replace("{target}", &"n".repeat(MAX_NICKNAME_ALIAS_LENGTH));
                            if expanded.chars().count() > MAX_MESSAGE_CHARACTERS {
                                return Err(ChatError::Configuration(format!(
                                    "social action {action_id} target template exceeds {MAX_MESSAGE_CHARACTERS} characters after nickname expansion"
                                )));
                            }
                        }
                        (None, None) => {},
                        _ => return Err(ChatError::Configuration(
                            "target Social templates must provide both argument and no-argument forms".to_string()
                        )),
                    }
                    let expanded_no_arg = action
                        .template_no_arg
                        .replace("{nickname}", &"n".repeat(MAX_NICKNAME_ALIAS_LENGTH));
                    if expanded_no_arg.chars().count() > MAX_MESSAGE_CHARACTERS {
                        return Err(ChatError::Configuration(format!(
                            "social action {action_id} no-argument template exceeds {MAX_MESSAGE_CHARACTERS} characters after nickname expansion"
                        )));
                    }
                    actions.insert(
                        action_id,
                        SocialActionConfig {
                            command: action.command,
                            aliases: action.aliases,
                            template: action.template,
                            template_no_arg: action.template_no_arg,
                            template_target: action.template_target,
                            template_target_no_arg: action.template_target_no_arg,
                        },
                    );
                }
                actions
            }
            _ => {
                return Err(ChatError::Configuration(
                    "chat command config version must be 1 or 2".to_string(),
                ));
            }
        };

        if raw.commands.is_empty() {
            return Err(ChatError::Configuration(
                "chat command config commands must not be empty".to_string(),
            ));
        }

        let max_nickname = "n".repeat(MAX_NICKNAME_ALIAS_LENGTH);
        for (command, response) in &raw.commands {
            let command_length = command.chars().count();
            if command_length < 2
                || command_length > MAX_MESSAGE_CHARACTERS
                || !command.starts_with('/')
                || command
                    .chars()
                    .any(|character| character.is_whitespace() || character.is_control())
            {
                return Err(ChatError::Configuration(
                    "chat command config keys must be slash-prefixed commands without whitespace"
                        .to_string(),
                ));
            }
            if response.trim().is_empty() {
                return Err(ChatError::Configuration(
                    "chat command config responses must not be empty".to_string(),
                ));
            }
            if response.chars().any(char::is_control) {
                return Err(ChatError::Configuration(
                    "chat command config responses must not contain control characters".to_string(),
                ));
            }
            let max_expanded_response = response.replace("{nickname}", &max_nickname);
            if max_expanded_response.chars().count() > MAX_MESSAGE_CHARACTERS {
                return Err(ChatError::Configuration(format!(
                    "chat command config responses must expand to at most {MAX_MESSAGE_CHARACTERS} characters"
                )));
            }
        }

        Ok(Self {
            commands: raw.commands,
            actions,
        })
    }

    pub fn expand(&self, command: &str, nickname: &str) -> Option<String> {
        self.commands
            .get(command)
            .map(|response| response.replace("{nickname}", nickname))
    }

    pub fn len(&self) -> usize {
        self.commands.len()
    }

    pub fn entries(&self) -> Vec<(String, String)> {
        let mut entries = self
            .commands
            .iter()
            .map(|(command, response)| (command.clone(), response.clone()))
            .collect::<Vec<_>>();
        entries.sort_by(|left, right| left.0.cmp(&right.0));
        entries
    }
}

pub(super) fn trusted_proxy_ips_from_env() -> Result<HashSet<IpAddr>, ChatError> {
    let value = match std::env::var("TRUSTED_PROXY_IPS") {
        Ok(value) => value,
        Err(std::env::VarError::NotPresent) => String::new(),
        Err(error) => {
            return Err(ChatError::Configuration(format!(
                "TRUSTED_PROXY_IPS could not be read: {error}"
            )));
        }
    };

    value
        .split(',')
        .map(str::trim)
        .filter(|value| !value.is_empty())
        .map(|value| {
            value.parse::<IpAddr>().map_err(|error| {
                ChatError::Configuration(format!(
                    "TRUSTED_PROXY_IPS contains an invalid IP address: {error}"
                ))
            })
        })
        .collect()
}

pub const MAX_MESSAGE_CHARACTERS: usize = 500;
pub const MAX_MESSAGE_FRAME_BYTES: usize = 4 * 1024;

#[derive(Deserialize)]
#[serde(deny_unknown_fields)]
struct IncomingMessage {
    #[serde(rename = "type")]
    kind: String,
    text: String,
}

#[derive(Clone, Debug, Deserialize, PartialEq, Eq, Serialize)]
pub struct ChatMessage {
    #[serde(rename = "type")]
    pub kind: String,
    pub id: String,
    pub nickname: String,
    pub text: String,
    #[serde(default = "default_message_kind")]
    pub message_kind: String,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub action_id: Option<String>,
    pub sent_at: String,
}

pub fn validate_message_frame(frame: &[u8]) -> Result<String, ChatError> {
    if frame.len() > MAX_MESSAGE_FRAME_BYTES {
        return Err(ChatError::FrameTooLarge);
    }

    let message: IncomingMessage =
        serde_json::from_slice(frame).map_err(|_| ChatError::InvalidMessage)?;
    if message.kind != "send" {
        return Err(ChatError::InvalidMessage);
    }

    let text = message.text.trim();
    if text.is_empty() {
        return Err(ChatError::InvalidMessage);
    }
    if text.chars().count() > MAX_MESSAGE_CHARACTERS {
        return Err(ChatError::MessageTooLong);
    }
    if text.chars().any(char::is_control) {
        return Err(ChatError::InvalidMessage);
    }

    Ok(text.to_string())
}

fn default_message_kind() -> String {
    "chat".to_string()
}

pub fn create_message(nickname: &str, text: String, action_id: Option<String>) -> ChatMessage {
    let mut bytes = [0_u8; 16];
    rand::rng().fill(&mut bytes);
    let id = bytes
        .iter()
        .map(|byte| format!("{byte:02x}"))
        .collect::<String>();

    ChatMessage {
        kind: "message".to_string(),
        id,
        nickname: nickname.to_string(),
        text,
        message_kind: if action_id.is_some() {
            "social"
        } else {
            "chat"
        }
        .to_string(),
        action_id,
        sent_at: chrono::Utc::now().to_rfc3339_opts(chrono::SecondsFormat::Millis, true),
    }
}

    fn issue_ticket_at_with_date(
        &self,
        client_ip: IpAddr,
        now: Instant,
        kst_date: Option<NaiveDate>,
    ) -> Result<ChatTicketResponse, ChatError> {
        let mut state = self.lock_state()?;
        prune_expired(&mut state, now);
        if !state.ip_limits.contains_key(&client_ip)
            && state.ip_limits.len() >= MAX_TRACKED_CLIENT_IPS
        {
            return Err(ChatError::RateLimited);
        }

        {
            let limits = state.ip_limits.entry(client_ip).or_default();
            prune_window(&mut limits.ticket_attempts, now, TICKET_REQUEST_WINDOW);
            if limits.ticket_attempts.len() >= MAX_TICKETS_PER_IP_PER_WINDOW {
                return Err(ChatError::RateLimited);
            }
            if limits.pending_tickets + limits.active_connections >= MAX_PENDING_AND_ACTIVE_PER_IP {
                return Err(ChatError::ConnectionLimitReached);
            }
            limits.ticket_attempts.push_back(now);
        }

        let kst_date = kst_date.unwrap_or_else(current_kst_date);
        if state.nickname_sequence_date != Some(kst_date) {
            state.nickname_sequences.clear();
            state.nickname_sequence_date = Some(kst_date);
        }

        let alias = {
            let ChatState {
                nickname_sequences,
                active_aliases,
                ..
            } = &mut *state;
            select_alias(&self.inner.nicknames, nickname_sequences, active_aliases)?
        };
        let token = generate_ticket(&state.tickets);
        state.active_aliases.insert(alias.clone());
        state
            .ip_limits
            .entry(client_ip)
            .or_default()
            .pending_tickets += 1;
        state.tickets.insert(
            token.clone(),
            TicketRecord {
                expires_at: now + TICKET_TTL,
                client_ip,
                alias: alias.clone(),
            },
        );

        Ok(ChatTicketResponse {
            ticket: token,
            nickname: alias,
            expires_in_seconds: TICKET_TTL.as_secs(),
        })
    }

    pub fn consume_ticket(&self, ticket: &str) -> Result<ChatConnection, ChatError> {
        self.consume_ticket_at(ticket, Instant::now())
    }

    fn consume_ticket_at(&self, ticket: &str, now: Instant) -> Result<ChatConnection, ChatError> {
        let mut state = self.lock_state()?;
        let ticket_record = state.tickets.remove(ticket);
        prune_expired(&mut state, now);

        let Some(record) = ticket_record else {
            return Err(ChatError::InvalidTicket);
        };
        if record.expires_at <= now {
            release_pending_ticket(&mut state, &record);
            prune_ip_limits(&mut state);
            return Err(ChatError::ExpiredTicket);
        }

        let Some(connection_id) = state.next_connection_id.checked_add(1) else {
            release_pending_ticket(&mut state, &record);
            prune_ip_limits(&mut state);
            return Err(ChatError::ServiceUnavailable);
        };
        state.next_connection_id = connection_id;
        if let Some(limits) = state.ip_limits.get_mut(&record.client_ip) {
            limits.pending_tickets = limits.pending_tickets.saturating_sub(1);
            limits.active_connections += 1;
        } else {
            release_alias(&mut state, &record.alias);
            return Err(ChatError::ServiceUnavailable);
        }
        state.connections.insert(
            connection_id,
            ConnectionRecord {
                client_ip: record.client_ip,
                alias: record.alias.clone(),
                message_attempts: VecDeque::new(),
            },
        );
        let online_count = state.connections.len();
        let _ = self.inner.presence.send(online_count);

        Ok(ChatConnection {
            inner: Arc::downgrade(&self.inner),
            id: connection_id,
            alias: record.alias,
        })
    }

    pub fn resolve_client_ip(
        &self,
        peer_ip: IpAddr,
        forwarded_for: Option<&str>,
    ) -> Result<IpAddr, ChatError> {
        if !self.inner.trusted_proxies.contains(&peer_ip) {
            return Ok(peer_ip);
        }

        let forwarded_for = forwarded_for.ok_or(ChatError::InvalidClientAddress)?;
        forwarded_for
            .split(',')
            .next()
            .map(str::trim)
            .filter(|value| !value.is_empty())
            .and_then(|value| value.parse::<IpAddr>().ok())
            .ok_or(ChatError::InvalidClientAddress)
    }

    pub fn cleanup_expired(&self) -> Result<(), ChatError> {
        let mut state = self.lock_state()?;
        prune_expired(&mut state, Instant::now());
        Ok(())
    }

    pub fn subscribe_messages(&self) -> broadcast::Receiver<ChatMessage> {
        self.inner.messages.subscribe()
    }

    pub fn subscribe_presence(&self) -> broadcast::Receiver<usize> {
        self.inner.presence.subscribe()
    }

    pub fn online_count(&self) -> usize {
        self.inner
            .state
            .lock()
            .map(|state| state.connections.len())
            .unwrap_or(0)
    }

    pub fn list_commands(&self) -> Vec<ChatCommandDefinition> {
        self.inner
            .commands
            .entries()
            .into_iter()
            .map(|(command, response)| ChatCommandDefinition { command, response })
            .collect()
    }

    pub fn list_actions(&self) -> Vec<ChatActionDefinition> {
        let mut actions = self
            .inner
            .commands
            .actions
            .iter()
            .map(|(action_id, action)| ChatActionDefinition {
                action_id: action_id.clone(),
                command: action.command.clone(),
                aliases: action.aliases.clone(),
                example: action.template_no_arg.replace("{nickname}", "누군가"),
            })
            .collect::<Vec<_>>();
        actions.sort_by(|left, right| left.command.cmp(&right.command));
        actions
    }

    pub fn publish_message(
        &self,
        connection: &ChatConnection,
        text: String,
    ) -> Result<(), ChatError> {
        connection.check_message_rate_limit()?;
        let (text, action_id) = if let Some(expanded) =
            self.inner.commands.expand(&text, connection.nickname())
        {
            (expanded, None)
        } else if let Some(matched) = parse_social_action(&text, &self.inner.commands.actions) {
            let action = self
                .inner
                .commands
                .actions
                .get(&matched.action_id)
                .ok_or(ChatError::ServiceUnavailable)?;
            let target_match =
                if action.template_target.is_some() && action.template_target_no_arg.is_some() {
                    matched.argument.as_deref().and_then(|argument| {
                        let state = self.inner.state.lock().ok()?;
                        state.connections.values().find_map(|online| {
                            let remainder = argument.strip_prefix(&online.alias)?;
                            if remainder.is_empty() {
                                Some((online.alias.clone(), None))
                            } else if remainder.chars().next().is_some_and(char::is_whitespace) {
                                let rest = remainder.trim_start();
                                Some((
                                    online.alias.clone(),
                                    (!rest.is_empty()).then(|| rest.to_string()),
                                ))
                            } else {
                                None
                            }
                        })
                    })
                } else {
                    None
                };
            let rendered = if let Some((target, argument)) = target_match {
                render_target_social_action(
                    action,
                    connection.nickname(),
                    &target,
                    argument.as_deref(),
                )?
            } else {
                render_social_action(action, connection.nickname(), matched.argument.as_deref())?
            };
            (rendered, Some(matched.action_id))
        } else {
            (text, None)
        };
        let message = create_message(connection.nickname(), text, action_id);
        self.inner
            .messages
            .send(message)
            .map(|_| ())
            .map_err(|_| ChatError::ServiceUnavailable)
    }

    pub fn spawn_cleanup_task(&self) {
        let service = self.clone();
        tokio::spawn(async move {
            let mut interval = tokio::time::interval(CLEANUP_INTERVAL);
            loop {
                interval.tick().await;
                if let Err(error) = service.cleanup_expired() {
                    tracing::error!(error = %error, "chat ticket cleanup failed");
                }
            }
        });
    }

    fn lock_state(&self) -> Result<std::sync::MutexGuard<'_, ChatState>, ChatError> {
        self.inner
            .state
            .lock()
            .map_err(|_| ChatError::ServiceUnavailable)
    }
}

impl ChatConnection {
    pub fn nickname(&self) -> &str {
        &self.alias
    }

    pub fn check_message_rate_limit(&self) -> Result<(), ChatError> {
        let inner = self.inner.upgrade().ok_or(ChatError::ServiceUnavailable)?;
        let now = Instant::now();
        let mut state = inner
            .state
            .lock()
            .map_err(|_| ChatError::ServiceUnavailable)?;

        let connection = state
            .connections
            .get_mut(&self.id)
            .ok_or(ChatError::InvalidTicket)?;
        prune_window(&mut connection.message_attempts, now, MESSAGE_RATE_WINDOW);
        if connection.message_attempts.len() >= MAX_MESSAGES_PER_CONNECTION_PER_WINDOW {
            return Err(ChatError::RateLimited);
        }
        connection.message_attempts.push_back(now);
        Ok(())
    }
}

impl Drop for ChatConnection {
    fn drop(&mut self) {
        let Some(inner) = self.inner.upgrade() else {
            return;
        };
        let mut state = match inner.state.lock() {
            Ok(state) => state,
            Err(poisoned) => poisoned.into_inner(),
        };
        let removed = if let Some(connection) = state.connections.remove(&self.id) {
            release_alias(&mut state, &self.alias);
            if let Some(limits) = state.ip_limits.get_mut(&connection.client_ip) {
                limits.active_connections = limits.active_connections.saturating_sub(1);
            }
            true
        } else {
            false
        };
        prune_ip_limits(&mut state);
        let online_count = state.connections.len();
        drop(state);
        if removed {
            let _ = inner.presence.send(online_count);
        }
    }
}

fn select_alias(
    config: &NicknameConfig,
    nickname_sequences: &mut HashMap<String, u64>,
    active_aliases: &HashSet<String>,
) -> Result<String, ChatError> {
    let word_index = rand::rng().random_range(0..config.words.len());
    next_alias_for_word(
        nickname_sequences,
        active_aliases,
        &config.words[word_index],
    )
}

fn next_alias_for_word(
    nickname_sequences: &mut HashMap<String, u64>,
    active_aliases: &HashSet<String>,
    word: &str,
) -> Result<String, ChatError> {
    let sequence = nickname_sequences.entry(word.to_string()).or_default();
    loop {
        *sequence = sequence
            .checked_add(1)
            .ok_or(ChatError::NicknameSequenceExhausted)?;
        let alias = format!("{word}-{sequence}");
        if !active_aliases.contains(&alias) {
            return Ok(alias);
        }
    }
}

fn current_kst_date() -> NaiveDate {
    kst_date_at(Utc::now())
}

fn kst_date_at(utc_time: chrono::DateTime<Utc>) -> NaiveDate {
    (utc_time + chrono::Duration::hours(9)).date_naive()
}

fn generate_ticket(existing: &HashMap<String, TicketRecord>) -> String {
    const HEX: &[u8; 16] = b"0123456789abcdef";
    let mut rng = rand::rng();

    loop {
        let mut bytes = [0_u8; TICKET_BYTES];
        rng.fill(&mut bytes);
        let mut token = String::with_capacity(TICKET_BYTES * 2);
        for byte in bytes {
            token.push(HEX[(byte >> 4) as usize] as char);
            token.push(HEX[(byte & 0x0f) as usize] as char);
        }
        if !existing.contains_key(&token) {
            return token;
        }
    }
}

fn prune_expired(state: &mut ChatState, now: Instant) {
    let expired_tickets = state
        .tickets
        .iter()
        .filter_map(|(ticket, record)| (record.expires_at <= now).then(|| ticket.clone()))
        .collect::<Vec<_>>();
    for ticket in expired_tickets {
        if let Some(record) = state.tickets.remove(&ticket) {
            release_pending_ticket(state, &record);
        }
    }

    for limits in state.ip_limits.values_mut() {
        prune_window(&mut limits.ticket_attempts, now, TICKET_REQUEST_WINDOW);
    }
    for connection in state.connections.values_mut() {
        prune_window(&mut connection.message_attempts, now, MESSAGE_RATE_WINDOW);
    }
    prune_ip_limits(state);
}

fn prune_window(queue: &mut VecDeque<Instant>, now: Instant, window: Duration) {
    while queue
        .front()
        .is_some_and(|timestamp| now.duration_since(*timestamp) >= window)
    {
        queue.pop_front();
    }
}

fn release_pending_ticket(state: &mut ChatState, record: &TicketRecord) {
    if let Some(limits) = state.ip_limits.get_mut(&record.client_ip) {
        limits.pending_tickets = limits.pending_tickets.saturating_sub(1);
    }
    release_alias(state, &record.alias);
}

fn release_alias(state: &mut ChatState, alias: &str) {
    state.active_aliases.remove(alias);
}

fn prune_ip_limits(state: &mut ChatState) {
    state.ip_limits.retain(|_, limits| {
        limits.pending_tickets > 0
            || limits.active_connections > 0
            || !limits.ticket_attempts.is_empty()
    });
}

use super::{ChatError, config::SocialActionConfig, message::MAX_MESSAGE_CHARACTERS};
use std::collections::HashMap;

/// A successful match against the final whitespace-delimited input token.
///
/// The argument is the complete input prefix before that token. Internal
/// whitespace and punctuation are preserved; boundary whitespace is removed.
#[derive(Clone, Debug, PartialEq, Eq)]
pub(super) struct MatchedSocialAction {
    pub(super) action_id: String,
    pub(super) matched_token: String,
    pub(super) argument: Option<String>,
}

/// Matches only the final whitespace-delimited token against registered
/// commands and aliases. Earlier tokens are never searched.
///
/// Callers pass normalized chat text (the WebSocket message validator trims
/// leading and trailing whitespace before this function is reached).
pub(super) fn parse_social_action(
    input: &str,
    actions: &HashMap<String, SocialActionConfig>,
) -> Option<MatchedSocialAction> {
    if input.is_empty() {
        return None;
    }

    let separator = input
        .char_indices()
        .rev()
        .find(|(_, character)| character.is_whitespace())
        .map(|(index, character)| (index, character.len_utf8()));

    let (argument, final_token) = match separator {
        Some((index, width)) => {
            let token_start = index + width;
            if token_start == input.len() {
                return None;
            }
            let argument = input[..index].trim();
            let argument = (!argument.is_empty()).then(|| argument.to_string());
            (argument, &input[token_start..])
        }
        None => (None, input),
    };

    for (action_id, action) in actions {
        let is_match = action.command == final_token
            || action.aliases.iter().any(|alias| alias == final_token);
        if is_match {
            return Some(MatchedSocialAction {
                action_id: action_id.clone(),
                matched_token: final_token.to_string(),
                argument,
            });
        }
    }
    None
}

/// Renders a matched social action without changing the parser's input semantics.
/// The argument placeholder receives one separating space when an argument exists.
pub(super) fn render_social_action(
    action: &SocialActionConfig,
    nickname: &str,
    argument: Option<&str>,
) -> Result<String, ChatError> {
    let rendered = match argument {
        Some(argument) if !argument.is_empty() => action
            .template
            .replace("{nickname}", nickname)
            .replace("{argument}", &format!(" {argument}")),
        _ => action.template_no_arg.replace("{nickname}", nickname),
    };

    if rendered.chars().any(char::is_control) {
        return Err(ChatError::InvalidMessage);
    }
    if rendered.chars().count() > MAX_MESSAGE_CHARACTERS {
        return Err(ChatError::MessageTooLong);
    }

    Ok(rendered)
}

pub(super) fn render_target_social_action(
    action: &SocialActionConfig,
    nickname: &str,
    target: &str,
    argument: Option<&str>,
) -> Result<String, ChatError> {
    let argument = argument.filter(|value| !value.is_empty());
    let template = match argument {
        Some(_) => action
            .template_target
            .as_deref()
            .ok_or(ChatError::ServiceUnavailable)?,
        None => action
            .template_target_no_arg
            .as_deref()
            .ok_or(ChatError::ServiceUnavailable)?,
    };
    let argument_text = argument
        .map(|value| format!(" {value}"))
        .unwrap_or_default();
    let rendered = template
        .replace("{nickname}", nickname)
        .replace("{target}", target)
        .replace("{argument}", &argument_text);
    if rendered.chars().any(char::is_control) {
        return Err(ChatError::InvalidMessage);
    }
    if rendered.chars().count() > MAX_MESSAGE_CHARACTERS {
        return Err(ChatError::MessageTooLong);
    }
    Ok(rendered)
}

// ============================================================
// 파일명: core/candles/aggregation.rs
// 설명: 1분봉에서 N분봉 파생 및 시간 그룹 계산
// ============================================================

use super::types::Candle;
use std::collections::BTreeMap;

/// 1분봉을 N분봉으로 파생
/// 그룹 키 = (날짜, 분 / N)
pub fn aggregate(candles: &[Candle], n: u32) -> Vec<Candle> {
    if n <= 1 || candles.is_empty() {
        return candles.to_vec();
    }

    // 그룹 키 → 병합 중인 캔들 (BTreeMap으로 시간순 유지)
    let mut groups: BTreeMap<u32, Candle> = BTreeMap::new();

    for c in candles {
        let total_min = parse_hhmm_to_minutes(&c.time);
        let group_key = total_min / n;

        groups
            .entry(group_key)
            .and_modify(|agg| {
                if c.high > agg.high {
                    agg.high = c.high;
                }
                if c.low < agg.low {
                    agg.low = c.low;
                }
                agg.close = c.close;
                agg.volume += c.volume;
            })
            .or_insert_with(|| c.clone());
    }

    groups.into_values().collect()
}

/// "HHMM" → 분 단위 숫자 (예: "0930" → 570)
pub(super) fn parse_hhmm_to_minutes(hhmm: &str) -> u32 {
    if hhmm.len() < 4 {
        return 0;
    }
    let h: u32 = hhmm[0..2].parse().unwrap_or(0);
    let m: u32 = hhmm[2..4].parse().unwrap_or(0);
    h * 60 + m
}

use super::{
    aggregation::{aggregate, parse_hhmm_to_minutes},
    types::{Candle, INTERVALS, interval_to_minutes},
};
use std::collections::{BTreeMap, HashMap};

fn is_valid_tick_time(value: &str) -> bool {
    if value.len() != 6 || !value.bytes().all(|byte| byte.is_ascii_digit()) {
        return false;
    }

    let hour = value[0..2].parse::<u8>().unwrap_or(u8::MAX);
    let minute = value[2..4].parse::<u8>().unwrap_or(u8::MAX);
    let second = value[4..6].parse::<u8>().unwrap_or(u8::MAX);

    hour <= 29 && minute <= 59 && second <= 59
}

fn is_valid_tick_date(value: &str) -> bool {
    value.len() == 8 && value.bytes().all(|byte| byte.is_ascii_digit())
}

pub struct CandleStore {
    // 종목코드 → 날짜("YYYY-MM-DD") → interval → 봉 배열
    data: HashMap<String, BTreeMap<String, HashMap<String, Vec<Candle>>>>,
}

impl CandleStore {
    pub fn new() -> Self {
        Self {
            data: HashMap::new(),
        }
    }

    pub fn load_1m(&mut self, code: &str, date: &str, candles_1m: Vec<Candle>) {
        // 주간(F)과 야간(CM) 조회 결과가 같은 영업일에 들어오므로
        // 같은 날짜 데이터는 시각 기준으로 병합하고 중복 분은 최신 입력으로 교체합니다.
        let existing = self
            .data
            .get(code)
            .and_then(|dates| dates.get(date))
            .and_then(|intervals| intervals.get("1m"))
            .cloned()
            .unwrap_or_default();
        let mut merged: BTreeMap<String, Candle> = existing
            .into_iter()
            .map(|candle| (candle.time.clone(), candle))
            .collect();
        for candle in candles_1m {
            merged.insert(candle.time.clone(), candle);
        }
        let merged_1m: Vec<Candle> = merged.into_values().collect();

        let mut per_interval: HashMap<String, Vec<Candle>> = HashMap::new();
        per_interval.insert("1m".to_string(), merged_1m.clone());

        for &interval in INTERVALS {
            if interval == "1m" {
                continue;
            }
            let n = match interval_to_minutes(interval) {
                Some(v) => v,
                None => continue,
            };
            per_interval.insert(interval.to_string(), aggregate(&merged_1m, n));
        }

        self.data
            .entry(code.to_string())
            .or_default()
            .insert(date.to_string(), per_interval);

        if let Some(dates) = self.data.get_mut(code) {
            while dates.len() > 2 {
                if let Some(oldest) = dates.keys().next().cloned() {
                    dates.remove(&oldest);
                }
            }
        }

        tracing::info!(
            "[CandleStore] {} {} 병합 로드: 1m={}건, 5m={}건, 60m={}건",
            code,
            date,
            self.count(code, date, "1m"),
            self.count(code, date, "5m"),
            self.count(code, date, "60m"),
        );
    }

    /// 해당 종목에 저장된 거래일을 오래된 순서로 반환합니다.
    pub fn dates(&self, code: &str) -> Vec<String> {
        self.data
            .get(code)
            .map(|dates| dates.keys().cloned().collect())
            .unwrap_or_default()
    }

    pub fn get(&self, code: &str, date: &str, interval: &str) -> Vec<Candle> {
        self.data
            .get(code)
            .and_then(|dates| dates.get(date))
            .and_then(|intervals| intervals.get(interval))
            .cloned()
            .unwrap_or_default()
    }

    pub fn count(&self, code: &str, date: &str, interval: &str) -> usize {
        self.data
            .get(code)
            .and_then(|dates| dates.get(date))
            .and_then(|intervals| intervals.get(interval))
            .map(|v| v.len())
            .unwrap_or(0)
    }

    /// WebSocket 체결 수신 → 모든 interval 봉 실시간 갱신
    /// - date: "YYYYMMDD" (예: "20261114")
    /// - hhmmss: "HHMMSS" (예: "093015")
    pub fn on_tick(&mut self, code: &str, date: &str, hhmmss: &str, price: f64, qty: u64) {
        if code.trim().is_empty()
            || !is_valid_tick_date(date)
            || !is_valid_tick_time(hhmmss)
            || !price.is_finite()
            || price <= 0.0
        {
            return;
        }
        let hhmm = &hhmmss[0..4]; // "HHMM"
        let hh: u32 = hhmm[0..2].parse().unwrap_or(0);
        let mm: u32 = hhmm[2..4].parse().unwrap_or(0);
        let total_min = hh * 60 + mm;

        for &interval in INTERVALS {
            let n = match interval_to_minutes(interval) {
                Some(v) => v,
                None => continue,
            };
            let group_key = total_min / n;

            let per_date = self
                .data
                .entry(code.to_string())
                .or_default()
                .entry(date.to_string())
                .or_default();

            let candles = per_date.entry(interval.to_string()).or_default();

            // 마지막 봉이 같은 그룹이면 갱신, 아니면 새 봉 시작
            let need_new = candles
                .last()
                .map(|c| parse_hhmm_to_minutes(&c.time) / n != group_key)
                .unwrap_or(true);

            if need_new {
                candles.push(Candle {
                    time: hhmm.to_string(),
                    open: price,
                    high: price,
                    low: price,
                    close: price,
                    volume: qty,
                });
            } else if let Some(last) = candles.last_mut() {
                if price > last.high {
                    last.high = price;
                }
                if price < last.low {
                    last.low = price;
                }
                last.close = price;
                last.volume += qty;
            }
        }

        // 2일치 초과 시 가장 오래된 날짜 제거
        if let Some(dates) = self.data.get_mut(code) {
            while dates.len() > 2 {
                if let Some(oldest) = dates.keys().next().cloned() {
                    dates.remove(&oldest);
                }
            }
        }
    }
}

// 설명: Supabase 오래된 데이터 정리 스케줄러
//       - 매일 KST 07:00에 30일 지난 데이터 삭제

use crate::core::db::SupabaseClient;

/// 매일 KST 07:00에 Supabase의 오래된 데이터를 정리하는 백그라운드 작업을 시작합니다.
pub fn spawn_daily_cleanup(supabase: SupabaseClient) {
    tokio::spawn(async move {
        use chrono::TimeZone;
        let kst = chrono_tz::Asia::Seoul;
        loop {
            let now = chrono::Utc::now().with_timezone(&kst);
            let Some(today_7am) = now.date_naive().and_hms_opt(7, 0, 0) else {
                tracing::error!("KST 데이터 정리 기준 시각 계산 실패");
                tokio::time::sleep(std::time::Duration::from_secs(60)).await;
                continue;
            };
            let Some(seven_am) = chrono::NaiveTime::from_hms_opt(7, 0, 0) else {
                tracing::error!("KST 데이터 정리 시각 생성 실패");
                tokio::time::sleep(std::time::Duration::from_secs(60)).await;
                continue;
            };
            let target_local = if now.time() < seven_am {
                Some(today_7am)
            } else {
                today_7am.checked_add_signed(chrono::Duration::days(1))
            };
            let Some(target_local) = target_local else {
                tracing::error!("다음 데이터 정리 날짜 계산 실패");
                tokio::time::sleep(std::time::Duration::from_secs(60)).await;
                continue;
            };
            let Some(next_7am) = kst.from_local_datetime(&target_local).single() else {
                tracing::error!("다음 KST 데이터 정리 시각 변환 실패");
                tokio::time::sleep(std::time::Duration::from_secs(60)).await;
                continue;
            };
            let wait = (next_7am - now)
                .to_std()
                .unwrap_or(std::time::Duration::from_secs(60));

            tracing::info!(
                "다음 데이터 삭제 예정: {} ({}초 후)",
                next_7am,
                wait.as_secs()
            );
            tokio::time::sleep(wait).await;

            if let Err(e) = supabase.cleanup_old_data().await {
                tracing::warn!("오래된 데이터 삭제 실패: {}", e);
            } else {
                tracing::info!("30일 지난 데이터 삭제 완료");
            }
        }
    });
}

        for table in ["notes", "news_links"] {

            if !resp.status().is_success() {
                let status = resp.status();
                let body = resp

                    http_status = %status,

                );
                return Err(format!("Supabase {} 삭제 HTTP {}: {}", table, status, body));
            }

/// KIS 토큰, 최근월물 종목, 초기 캔들 저장소를 준비합니다.
pub async fn initialize(

    app_key: &str,
    app_secret: &str,
    cache_path: &str,
) -> Result<MarketData, Box<dyn std::error::Error>> {
    // KIS 토큰
    let token = match kis::get_or_issue_token(http_client, app_key, app_secret, cache_path).await {

        }
    };

        Err(e) => {

            return Err(std::io::Error::new(
                std::io::ErrorKind::Other,
                "마스터 파일 다운로드 실패",
            )
            .into());
        }
    };
    let regular = match master::pick_nearest(&symbols, "1") {
        Some(f) => {
            tracing::info!("정규선물 최근월물: {} ({})", f.name, f.code);
            f
        }
        None => {
            tracing::error!("정규선물 최근월물 없음");
            return Err(std::io::Error::new(
                std::io::ErrorKind::NotFound,
                "정규선물 최근월물 없음",
            )
            .into());
        }
    };
    let mini = match master::pick_nearest(&symbols, "B") {
        Some(f) => {
            tracing::info!("미니선물 최근월물: {} ({})", f.name, f.code);

        }

                std::io::ErrorKind::NotFound,
                "미니선물 최근월물 없음",
            )

        regular,
        mini,

    })

//! KST 기준 KIS 선물 세션 구독 시간 판정.
use chrono::NaiveTime;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct ActiveSessions {
    pub day: bool,
    pub night: bool,
}

pub fn active_sessions_at(time: NaiveTime) -> ActiveSessions {
    let day_start = NaiveTime::from_hms_opt(8, 40, 0).unwrap();
    let day_end = NaiveTime::from_hms_opt(15, 50, 0).unwrap();
    let night_start = NaiveTime::from_hms_opt(17, 55, 0).unwrap();
    let night_end = NaiveTime::from_hms_opt(6, 5, 0).unwrap();

    ActiveSessions {
        day: time >= day_start && time < day_end,
        // 자정을 넘는 구간: 17:55 이상 또는 06:05 이전.
        night: time >= night_start || time < night_end,
    }
}

pub fn current_active_sessions() -> ActiveSessions {
    let now = chrono::Utc::now().with_timezone(&chrono_tz::Asia::Seoul);
    active_sessions_at(now.time())
}

use chrono::{DateTime, Duration, NaiveDate, TimeZone, Timelike, Utc};
use serde::Serialize;

const DAY_START: (u32, u32) = (8, 45);
const DAY_END: (u32, u32) = (9, 30);
const NIGHT_START: (u32, u32) = (18, 0);
const NIGHT_END: (u32, u32) = (18, 45);

#[derive(Debug, Clone, Serialize)]
pub struct RangeAlert {
    pub session: String,
    pub direction: String,
    pub price: f64,
    pub reference_price: f64,
    pub message: String,
}

#[derive(Debug, Clone, Serialize)]
pub struct SessionRangeSnapshot {
    pub session: String,
    pub date: Option<String>,
    pub status: String,
    pub high: Option<f64>,
    pub low: Option<f64>,
    pub high_alerted: bool,
    pub low_alerted: bool,
}

#[derive(Debug, Clone, Serialize)]
pub struct RangeMonitorSnapshot {
    pub day: SessionRangeSnapshot,
    pub night: SessionRangeSnapshot,
}

#[derive(Debug, Default)]
struct SessionState {
    date: Option<String>,
    eligible: bool,
    high: Option<f64>,
    low: Option<f64>,
    high_alerted: bool,
    low_alerted: bool,
}

impl SessionState {
    fn new(date: &str, eligible: bool) -> Self {
        Self {
            date: Some(date.to_string()),
            eligible,
            ..Self::default()
        }
    }
}

/// 서버에서 정규선물 틱만 받아 주간/야간 시초 레인지를 독립적으로 관리합니다.
pub struct SessionRangeMonitor {
    regular_code: String,
    /// 최초 KIS WebSocket 구독 완료 시각. 구독 준비 전에는 경보 자격이 없습니다.
    started_at: Option<DateTime<Utc>>,
    day: SessionState,
    night: SessionState,
}

impl SessionRangeMonitor {
    pub fn new(regular_code: String) -> Self {
        Self {
            regular_code,
            started_at: None,
            day: SessionState::default(),
            night: SessionState::default(),
        }
    }

    fn new_at(regular_code: String, started_at: DateTime<Utc>) -> Self {
        Self {
            regular_code,
            started_at: Some(started_at),
            day: SessionState::default(),
            night: SessionState::default(),
        }
    }

    /// KIS WebSocket 구독 요청이 완료된 첫 시각을 기록합니다. 재연결 때는 최초 시각을 유지합니다.
    pub fn mark_feed_ready(&mut self) {
        let now = Utc::now();
        if self.started_at.is_none() {
            self.started_at = Some(now);
        } else {
            self.invalidate_if_collecting(now);
        }
    }

    /// 기준 구간 중 WebSocket이 끊기면 불완전한 레인지를 폐기합니다.
    pub fn mark_feed_disconnected(&mut self) {
        self.invalidate_if_collecting(Utc::now());
    }

    fn invalidate_if_collecting(&mut self, now_utc: DateTime<Utc>) {
        let now = now_utc.with_timezone(&chrono_tz::Asia::Seoul);
        let date = now.date_naive().format("%Y%m%d").to_string();
        let hhmm = format!("{:02}{:02}", now.hour(), now.minute());

        if hhmm.as_str() >= "0845" && hhmm.as_str() < "0930" {
            if self.day.date.as_deref() != Some(date.as_str()) {
                self.day = SessionState::new(&date, true);
            }
            self.day.eligible = false;
            self.day.high = None;
            self.day.low = None;
        }

        if hhmm.as_str() >= "1800" && hhmm.as_str() < "1845" {
            if self.night.date.as_deref() != Some(date.as_str()) {
                self.night = SessionState::new(&date, true);
            }
            self.night.eligible = false;
            self.night.high = None;
            self.night.low = None;
        }
    }

    pub fn on_tick(
        &mut self,
        code: &str,
        date: &str,
        time: &str,
        market_session: &str,
        price: f64,
    ) -> Option<RangeAlert> {
        if code != self.regular_code || !price.is_finite() || price <= 0.0 {
            return None;
        }

        let started_at = self.started_at.clone();
        let (session, start, end, state) = match market_session {
            "day" => ("day", DAY_START, DAY_END, &mut self.day),
            "night" => ("night", NIGHT_START, NIGHT_END, &mut self.night),
            _ => return None,
        };

        if !valid_date(date) || !valid_time(time) {
            return None;
        }

        if state.date.as_deref() != Some(date) {
            let eligible = session_start_utc(date, start).is_some_and(|session_start| {
                started_at
                    .as_ref()
                    .is_some_and(|started| *started <= session_start)
            });
            *state = SessionState::new(date, eligible);
        }

        if !state.eligible {
            return None;
        }

        let hhmm = &time[..4];
        let start_hhmm = format!("{:02}{:02}", start.0, start.1);
        let end_hhmm = format!("{:02}{:02}", end.0, end.1);

        if hhmm < start_hhmm.as_str() {
            return None;
        }

        if hhmm < end_hhmm.as_str() {
            state.high = Some(state.high.map_or(price, |high| high.max(price)));
            state.low = Some(state.low.map_or(price, |low| low.min(price)));
            return None;
        }

        let (Some(high), Some(low)) = (state.high, state.low) else {
            return None;
        };

        if price > high && !state.high_alerted {
            state.high_alerted = true;
            return Some(RangeAlert {
                session: session.to_string(),
                direction: "high".to_string(),
                price,
                reference_price: high,
                message: "시세가 고가 돌파 했습니다. 숏 포지션은 위험 관리 하세요.".to_string(),
            });
        }

        if price < low && !state.low_alerted {
            state.low_alerted = true;
            return Some(RangeAlert {
                session: session.to_string(),
                direction: "low".to_string(),
                price,
                reference_price: low,
                message: "시세가 저가를 깼습니다. 롱 포지션은 위험 관리 하세요.".to_string(),
            });
        }

        None
    }

    pub fn snapshot(&self) -> RangeMonitorSnapshot {
        let now = Utc::now().with_timezone(&chrono_tz::Asia::Seoul);
        let today = now.date_naive();
        let day_date = today.format("%Y%m%d").to_string();
        let day_hhmm = format!("{:02}{:02}", now.hour(), now.minute());

        // KIS 야간 체결은 자정 이후 영업일 전날의 24시 이후 시각으로 정규화됩니다.
        let night_date = if now.hour() < 6 {
            (today - Duration::days(1)).format("%Y%m%d").to_string()
        } else {
            day_date.clone()
        };
        let night_hour = if now.hour() < 6 {
            now.hour() + 24
        } else {
            now.hour()
        };
        let night_hhmm = format!("{:02}{:02}", night_hour, now.minute());

        RangeMonitorSnapshot {
            day: self.snapshot_session(&self.day, "day", &day_date, &day_hhmm, DAY_START, DAY_END),
            night: self.snapshot_session(
                &self.night,
                "night",
                &night_date,
                &night_hhmm,
                NIGHT_START,
                NIGHT_END,
            ),
        }
    }

    fn snapshot_session(
        &self,
        state: &SessionState,
        session: &str,
        current_date: &str,
        current_hhmm: &str,
        start: (u32, u32),
        end: (u32, u32),
    ) -> SessionRangeSnapshot {
        let date = state.date.as_deref().unwrap_or(current_date);
        let eligible = state.eligible
            || (state.date.is_none()
                && session_start_utc(date, start).is_some_and(|session_start| {
                    self.started_at
                        .as_ref()
                        .is_some_and(|started| *started <= session_start)
                }));
        let start_hhmm = format!("{:02}{:02}", start.0, start.1);
        let end_hhmm = format!("{:02}{:02}", end.0, end.1);
        let status = if !eligible {
            "unavailable"
        } else if current_hhmm < start_hhmm.as_str() {
            "waiting"
        } else if current_hhmm < end_hhmm.as_str() {
            "collecting"
        } else if state.high.is_some() && state.low.is_some() {
            "monitoring"
        } else {
            "unavailable"
        };

        SessionRangeSnapshot {
            session: session.to_string(),
            date: Some(date.to_string()),
            status: status.to_string(),
            high: state.high,
            low: state.low,
            high_alerted: state.high_alerted,
            low_alerted: state.low_alerted,
        }
    }
}

fn valid_date(value: &str) -> bool {
    value.len() == 8
        && value.bytes().all(|byte| byte.is_ascii_digit())
        && NaiveDate::parse_from_str(value, "%Y%m%d").is_ok()
}

fn valid_time(value: &str) -> bool {
    if value.len() != 6 || !value.bytes().all(|byte| byte.is_ascii_digit()) {
        return false;
    }
    let hour = value[0..2].parse::<u32>().unwrap_or(u32::MAX);
    let minute = value[2..4].parse::<u32>().unwrap_or(u32::MAX);
    let second = value[4..6].parse::<u32>().unwrap_or(u32::MAX);
    hour <= 29 && minute <= 59 && second <= 59
}

fn session_start_utc(date: &str, start: (u32, u32)) -> Option<DateTime<Utc>> {
    let date = NaiveDate::parse_from_str(date, "%Y%m%d").ok()?;
    let local = date.and_hms_opt(start.0, start.1, 0)?;
    chrono_tz::Asia::Seoul
        .from_local_datetime(&local)
        .single()
        .map(|value| value.with_timezone(&Utc))
}

pub fn today_yyyymmdd(utc_now: DateTime<Utc>) -> String {

        .format("%Y%m%d")

    (utc_now.with_timezone(&chrono_tz::Asia::Seoul).date_naive() - Duration::days(1))
        .format("%Y%m%d")
        .to_string()
}

use serde::{Deserialize, Serialize};
use std::path::Path;

// ── REST API OAuth 토큰 ──

#[derive(Debug, Serialize)]
struct TokenRequest {
    grant_type: String,
    appkey: String,
    appsecret: String,
}

#[derive(Debug, Deserialize, Serialize, Clone)]
pub struct TokenResponse {
    pub access_token: String,
    pub token_type: String,
    pub expires_in: u64,
    pub access_token_token_expired: String,
}

#[derive(Debug, Serialize, Deserialize)]
struct CachedToken {
    token: TokenResponse,
    issued_at: u64,
}

pub async fn get_or_issue_token(
    client: &reqwest::Client,
    app_key: &str,
    app_secret: &str,
    cache_path: &str,
) -> Result<TokenResponse, Box<dyn std::error::Error>> {
    if Path::new(cache_path).exists() {
        if let Ok(content) = std::fs::read_to_string(cache_path) {
            if let Ok(cached) = serde_json::from_str::<CachedToken>(&content) {
                let now = std::time::SystemTime::now()
                    .duration_since(std::time::UNIX_EPOCH)?
                    .as_secs();
                if now.saturating_sub(cached.issued_at) < 23 * 3600 {
                    tracing::info!("KIS token loaded from cache");
                    return Ok(cached.token);
                }
            }
        }
    }

    let body = TokenRequest {
        grant_type: "client_credentials".to_string(),
        appkey: app_key.to_string(),
        appsecret: app_secret.to_string(),
    };

    let rest_domain = std::env::var("KIS_REST_DOMAIN")
        .unwrap_or_else(|_| "https://openapi.koreainvestment.com:9443".to_string());
    let url = format!("{}/oauth2/tokenP", rest_domain.trim_end_matches('/'));

    let token = client
        .post(&url)
        .header("Content-Type", "application/json; charset=UTF-8")
        .json(&body)
        .send()
        .await?
        .error_for_status()?
        .json::<TokenResponse>()
        .await?;

    let now = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)?
        .as_secs();
    let cached = CachedToken {
        token: token.clone(),
        issued_at: now,
    };
    std::fs::write(cache_path, serde_json::to_string(&cached)?)?;
    tracing::info!("KIS token issued and cached");

    Ok(token)
}

/// 시장 구분(F: 주간, CM: 야간)을 지정해 선물 1분봉을 조회합니다.
pub async fn get_minute_chart_for_market(
    client: &reqwest::Client,
    app_key: &str,
    app_secret: &str,
    access_token: &str,
    market_div_code: &str,
    symbol: &str,
    date: &str,
    hour: &str,
) -> Result<Vec<MinuteCandle>, Box<dyn std::error::Error + Send + Sync>> {
    let rest_domain = std::env::var("KIS_REST_DOMAIN")
        .unwrap_or_else(|_| "https://openapi.koreainvestment.com:9443".to_string());
    let url = format!(
        "{}/uapi/domestic-futureoption/v1/quotations/inquire-time-fuopchartprice",
        rest_domain.trim_end_matches('/')
    );

    let query = [
        ("FID_COND_MRKT_DIV_CODE", market_div_code),
        ("FID_INPUT_ISCD", symbol),
        ("FID_HOUR_CLS_CODE", "60"),
        ("FID_PW_DATA_INCU_YN", "Y"),
        ("FID_FAKE_TICK_INCU_YN", "N"),
        ("FID_INPUT_DATE_1", date),
        ("FID_INPUT_HOUR_1", hour),
    ];
    let request_started = std::time::Instant::now();
    tracing::info!(
        target: "kis::night_candles",
        "[MINUTE-REQUEST] market={} symbol={} date={} cursor={} tr_id=FHKIF03020200 url={} query={:?}",
        market_div_code, symbol, date, hour, url, query
    );

    let response = client
        .get(&url)
        .header("Content-Type", "application/json; charset=utf-8")
        .header("authorization", format!("Bearer {}", access_token))
        .header("appkey", app_key)
        .header("appsecret", app_secret)
        .header("tr_id", "FHKIF03020200")
        .header("custtype", "P")
        .query(&query)
        .send()
        .await?;
    let status = response.status();
    let response_headers = format!("{:?}", response.headers());
    let body = response.bytes().await?;
    let elapsed_ms = request_started.elapsed().as_millis();

    if !status.is_success() {
        let body_text = String::from_utf8_lossy(&body);
        tracing::error!(
            target: "kis::night_candles",
            "[MINUTE-HTTP-ERROR] market={} symbol={} date={} cursor={} status={} elapsed_ms={} headers={} body={}",
            market_div_code, symbol, date, hour, status, elapsed_ms, response_headers, body_text
        );
        return Err(std::io::Error::other(format!(
            "KIS minute chart HTTP {} (market={}, symbol={}, date={}, cursor={}): {}",
            status, market_div_code, symbol, date, hour, body_text
        ))
        .into());
    }

    tracing::info!(
        target: "kis::night_candles",
        "[MINUTE-HTTP-OK] market={} symbol={} date={} cursor={} status={} bytes={} elapsed_ms={} headers={}",
        market_div_code, symbol, date, hour, status, body.len(), elapsed_ms, response_headers
    );

    let resp: MinuteChartResponse = serde_json::from_slice(&body).map_err(|error| {
        let body_text = String::from_utf8_lossy(&body);
        tracing::error!(
            target: "kis::night_candles",
            "[MINUTE-JSON-ERROR] market={} symbol={} date={} cursor={} error={} body={}",
            market_div_code, symbol, date, hour, error, body_text
        );
        error
    })?;

    tracing::info!(
        target: "kis::night_candles",
        "[MINUTE-RESPONSE] market={} symbol={} date={} cursor={} rt_cd={} msg_cd={} msg1={} output2_count={}",
        market_div_code, symbol, date, hour, resp.rt_cd, resp.msg_cd, resp.msg1, resp.output2.len()
    );
    ensure_success(&resp).map_err(std::io::Error::other)?;

    let mut candles: Vec<MinuteCandle> = resp
        .output2
        .into_iter()
        .map(|c| MinuteCandle {
            stck_bsop_date: c.stck_bsop_date,
            stck_cntg_hour: c.stck_cntg_hour,
            futs_prpr: c.futs_prpr,
            futs_oprc: c.futs_oprc,
            futs_hgpr: c.futs_hgpr,
            futs_lwpr: c.futs_lwpr,
            cntg_vol: c.cntg_vol,
        })
        .collect();

    candles.sort_by(|a, b| {
        (a.stck_bsop_date.clone(), a.stck_cntg_hour.clone())
            .cmp(&(b.stck_bsop_date.clone(), b.stck_cntg_hour.clone()))
    });
    let first = candles
        .first()
        .map(|c| format!("{} {}", c.stck_bsop_date, c.stck_cntg_hour))
        .unwrap_or_else(|| "none".to_string());
    let last = candles
        .last()
        .map(|c| format!("{} {}", c.stck_bsop_date, c.stck_cntg_hour))
        .unwrap_or_else(|| "none".to_string());
    tracing::info!(
        target: "kis::night_candles",
        "[MINUTE-PARSED] market={} symbol={} date={} cursor={} candles={} first={} last={}",
        market_div_code, symbol, date, hour, candles.len(), first, last
    );
    Ok(candles)
}

async fn set_subscription<S>(

) -> Result<(), Box<dyn std::error::Error>>

// KIS 실시간 시세 프로토콜 해석

use crate::core::session_range::RangeAlert;
use crate::core::time::{previous_calendar_day_yyyymmdd, today_yyyymmdd};
use tokio::sync::mpsc;

pub(super) const DAY_TR_ID: &str = "H0IFCNT0";
pub(super) const NIGHT_TR_ID: &str = "H0MFCNT0";

// 주간/야간 선물 체결 프레임의 공통 필드 인덱스.
const FIELD_CODE: usize = 0;
const FIELD_TIME: usize = 1;
const FIELD_CHANGE: usize = 2;
const FIELD_CHANGE_SIGN: usize = 3;
const FIELD_CHANGE_RATE: usize = 4;
const FIELD_PRICE: usize = 5;
const FIELD_LAST_QTY: usize = 9;
const FIELD_CUMULATIVE_VOLUME: usize = 10;

fn is_valid_hhmmss(value: &str) -> bool {
    if value.len() != 6 || !value.bytes().all(|byte| byte.is_ascii_digit()) {
        return false;
    }
    let hour = value[0..2].parse::<u8>().unwrap_or(u8::MAX);
    let minute = value[2..4].parse::<u8>().unwrap_or(u8::MAX);
    let second = value[4..6].parse::<u8>().unwrap_or(u8::MAX);
    hour <= 29 && minute <= 59 && second <= 59
}

#[derive(Debug, Clone)]
pub struct RealtimeTick {
    pub code: String,
    #[allow(dead_code)]
    pub date: String,
    pub price: String,
    pub change: String,
    pub change_rate: String,
    pub volume: String,
    pub last_qty: String,
    pub time: String,
    /// 주간 체결은 "day", KRX 야간 체결은 "night".
    pub market_session: String,
    pub range_alert: Option<RangeAlert>,
}

/// 주간(H0IFCNT0) 및 KRX 야간선물(H0MFCNT0) 체결을 공통 Tick으로 변환합니다.
pub(super) fn parse_tick(text: &str) -> Option<RealtimeTick> {
    let mut parts = text.splitn(4, '|');
    let _encrypted = parts.next()?;
    let tr_id = parts.next()?;
    if tr_id != DAY_TR_ID && tr_id != NIGHT_TR_ID {
        return None;
    }
    let _count = parts.next()?;
    let fields: Vec<&str> = parts.next()?.split('^').collect();
    if fields.len() <= FIELD_CUMULATIVE_VOLUME
        || fields[FIELD_CODE].is_empty()
        || !is_valid_hhmmss(fields[FIELD_TIME])
    {
        return None;
    }

    let price = fields[FIELD_PRICE].parse::<f64>().ok()?;
    if !price.is_finite() || price <= 0.0 {
        return None;
    }
    let last_qty = fields[FIELD_LAST_QTY].parse::<u64>().ok()?;
    let cumulative_volume = fields[FIELD_CUMULATIVE_VOLUME].parse::<u64>().ok()?;

    let utc_now = chrono::Utc::now();
    let is_night = tr_id == NIGHT_TR_ID;
    let hour = fields[FIELD_TIME][0..2].parse::<u32>().ok()?;
    let (date, time) = if is_night && hour < 6 {
        // CM 분봉 API는 자정 이후를 전일 영업일의 24시 이후(예: 25:30)로 표현합니다.
        (
            previous_calendar_day_yyyymmdd(utc_now),
            format!("{:02}{}", hour + 24, &fields[FIELD_TIME][2..]),
        )
    } else {
        (today_yyyymmdd(utc_now), fields[FIELD_TIME].to_string())
    };

    Some(RealtimeTick {
        code: fields[FIELD_CODE].to_string(),
        date,
        time,
        change: format!("{}{}", fields[FIELD_CHANGE_SIGN], fields[FIELD_CHANGE]),
        change_rate: fields[FIELD_CHANGE_RATE].to_string(),
        price: fields[FIELD_PRICE].to_string(),
        last_qty: last_qty.to_string(),
        volume: cumulative_volume.to_string(),
        market_session: if is_night { "night" } else { "day" }.to_string(),
        range_alert: None,
    })
}

pub(super) fn handle_text(text: &str, pong_tx: &mpsc::Sender<String>) {
    if text.contains(r#""tr_id":"PINGPONG""#) {
        tracing::debug!("PINGPONG 수신, 응답 전송 예약");
        let _ = pong_tx.try_send(text.to_string());
        return;
    }

    let tr_id = text.split('|').nth(1).unwrap_or_default();
    if tr_id != DAY_TR_ID && tr_id != NIGHT_TR_ID {
        tracing::debug!("WS 응답: {}", text);
    }
}

/// 로깅 초기화. 반환된 guard는 프로그램 종료 시까지 유지해야 함.
pub fn init() -> tracing_appender::non_blocking::WorkerGuard {
    // logs 디렉토리 생성
    let log_dir = PathBuf::from(std::env::var("LOG_DIR").unwrap_or_else(|_| "./logs".to_string()));

    // 일별 로테이션 파일 appender: logs/2026-11-14.log
    let file_appender = RollingFileAppender::builder()
        .rotation(Rotation::DAILY)
        .filename_suffix("log")
        .build(&log_dir)
        .expect("로그 파일 appender 생성 실패");

    let (non_blocking, guard) = tracing_appender::non_blocking(file_appender);

    // 파일 레이어만 사용 (콘솔 출력 없음)
    let file_layer = fmt::layer()
        .with_writer(non_blocking)
        .with_ansi(false) // 파일에 색상 코드 없음
        .with_target(true)
        .with_line_number(true)
        .with_file(true);

    // 필터: LOG_LEVEL 없으면 info
    let filter = EnvFilter::try_from_env("LOG_LEVEL").unwrap_or_else(|_| EnvFilter::new("info"));

    tracing_subscriber::registry()
        .with(filter)
        .with(file_layer)
        .init();

    guard
}

/// 18시 이전에는 대기하고, 야간 세션 중 서버가 시작되면 현재 시각부터 18시까지 CM 분봉을 로드합니다.
/// 야간 체결 WebSocket 구독은 별도로 서버 시작 시 신청되어 있으므로 이후 봉은 실시간으로 누적됩니다.
pub fn spawn_night_candle_loader(
    http_client: reqwest::Client,
    app_key: String,
    app_secret: String,
    access_token: String,
    symbols: Vec<String>,
    candle_store: std::sync::Arc<tokio::sync::Mutex<CandleStore>>,
) {
    tokio::spawn(async move {
        tracing::info!(
            target: "kis::night_candles",
            "[NIGHT-LOADER-START] symbols={:?} page_size=102 max_pages=8 market=CM",
            symbols
        );

        loop {
            let now = chrono::Utc::now().with_timezone(&chrono_tz::Asia::Seoul);
            let hour = chrono::Timelike::hour(&now);
            let session_active = hour >= 18 || hour < 6;

            if !session_active {
                let next_start = now
                    .date_naive()
                    .and_hms_opt(18, 0, 0)
                    .expect("valid night session start");
                let wait = (next_start - now.naive_local())
                    .to_std()
                    .unwrap_or_default();
                tracing::info!(
                    target: "kis::night_candles",
                    "[NIGHT-WAIT] now={} next_session=18:00:00 wait_seconds={}",
                    now, wait.as_secs()
                );
                tokio::time::sleep(wait).await;
            }

            let session_now = chrono::Utc::now().with_timezone(&chrono_tz::Asia::Seoul);
            let session_hour = chrono::Timelike::hour(&session_now);
            let (trade_date, cursor_hour) = if session_hour < 6 {
                (
                    (session_now.date_naive() - chrono::Duration::days(1))
                        .format("%Y%m%d")
                        .to_string(),
                    format!("{:02}{}", session_hour + 24, session_now.format("%M%S")),
                )
            } else {
                (
                    session_now.format("%Y%m%d").to_string(),
                    session_now.format("%H%M%S").to_string(),
                )
            };

            tracing::info!(
                target: "kis::night_candles",
                "[NIGHT-SESSION-BEGIN] now={} trade_date={} initial_cursor={} symbols={}",
                session_now, trade_date, cursor_hour, symbols.len()
            );

            for symbol in &symbols {
                tracing::info!(
                    target: "kis::night_candles",
                    "[NIGHT-SYMBOL-BEGIN] symbol={} date={} cursor={}",
                    symbol, trade_date, cursor_hour
                );
                let load_result = load_night_session(
                    &http_client,
                    &app_key,
                    &app_secret,
                    &access_token,
                    symbol,
                    &trade_date,
                    &cursor_hour,
                )
                .await
                .map_err(|error| error.to_string());

                match load_result {
                    Ok(candles) => {
                        let count = candles.len();
                        tracing::info!(
                            target: "kis::night_candles",
                            "[NIGHT-SYMBOL-DONE] symbol={} date={} loaded_candles={}",
                            symbol, trade_date, count
                        );
                        if !candles.is_empty() {
                            let mut store = candle_store.lock().await;
                            store.load_1m(symbol, &trade_date, candles);
                            tracing::info!(
                                target: "kis::night_candles",
                                "[NIGHT-STORE-MERGED] symbol={} date={} loaded_candles={} store_merge=complete",
                                symbol, trade_date, count
                            );
                        } else {
                            tracing::warn!(
                                target: "kis::night_candles",
                                "[NIGHT-STORE-SKIP] symbol={} date={} reason=no_candles",
                                symbol, trade_date
                            );
                        }
                    }
                    Err(error) => tracing::error!(
                        target: "kis::night_candles",
                        "[NIGHT-SYMBOL-FAILED] symbol={} date={} initial_cursor={} error={}",
                        symbol, trade_date, cursor_hour, error
                    ),
                }
            }

            let after_load = chrono::Utc::now().with_timezone(&chrono_tz::Asia::Seoul);
            let next_start_date = if chrono::Timelike::hour(&after_load) < 6 {
                after_load.date_naive()
            } else {
                after_load.date_naive() + chrono::Duration::days(1)
            };
            let next_day_start = next_start_date
                .and_hms_opt(18, 0, 0)
                .expect("valid next night session start");
            let wait = (next_day_start - after_load.naive_local())
                .to_std()
                .unwrap_or_default();
            tracing::info!(
                target: "kis::night_candles",
                "[NIGHT-SESSION-END] now={} next_run={} wait_seconds={}",
                after_load, next_day_start, wait.as_secs()
            );
            tokio::time::sleep(wait).await;
        }
    });
}

async fn load_night_session(
    http_client: &reqwest::Client,
    app_key: &str,
    app_secret: &str,
    access_token: &str,
    symbol: &str,
    trade_date: &str,
    initial_hour: &str,
) -> Result<Vec<crate::core::candles::Candle>, Box<dyn std::error::Error>> {
    let mut result = Vec::new();
    let mut cursor = initial_hour.to_string();

    tracing::info!(
        target: "kis::night_candles",
        "[NIGHT-PAGINATION-START] symbol={} date={} start_cursor={} page_size=102 max_pages=8",
        symbol, trade_date, cursor
    );

    // KIS 분봉 조회의 초당 호출 제한을 피하기 위한 요청 간격입니다.
    // 기본값 300ms이며 KIS_CANDLE_REQUEST_DELAY_MS 환경변수로 조정할 수 있습니다.
    let request_delay_ms = std::env::var("KIS_CANDLE_REQUEST_DELAY_MS")
        .ok()
        .and_then(|value| value.parse::<u64>().ok())
        .unwrap_or(300);
    const MAX_RATE_LIMIT_RETRIES: usize = 3;

    for page in 0..8 {
        let page_started = std::time::Instant::now();
        tracing::info!(
            target: "kis::night_candles",
            "[NIGHT-PAGE-REQUEST] symbol={} date={} page={} cursor={} market=CM",
            symbol, trade_date, page + 1, cursor
        );

        // 호출 제한(EGW00201)이면 동일 cursor를 유지한 채 대기 후 재시도합니다.
        let mut retry = 0usize;
        let chunk = loop {
            match kis::get_minute_chart_for_market(
                http_client,
                app_key,
                app_secret,
                access_token,
                "CM",
                symbol,
                trade_date,
                &cursor,
            )
            .await
            {
                Ok(chunk) => {
                    // 성공 응답 후에도 다음 요청(다음 종목 포함) 전 간격을 확보합니다.
                    tokio::time::sleep(std::time::Duration::from_millis(request_delay_ms)).await;
                    break chunk;
                }
                Err(error) => {
                    // Box<dyn Error>는 Send를 보장하지 않으므로 await 전에 문자열로 변환합니다.
                    let error_message = error.to_string();
                    // 명시적으로 원본 Box<dyn Error>를 await 이전에 해제해야 Send 조건을 만족합니다.
                    drop(error);
                    if error_message.contains("EGW00201") && retry < MAX_RATE_LIMIT_RETRIES {
                        retry += 1;
                        let backoff_ms = request_delay_ms
                            .max(1000u64.saturating_mul(2u64.pow(retry as u32 - 1)));
                        tracing::warn!(
                            target: "kis::night_candles",
                            "[NIGHT-RATE-LIMIT-RETRY] symbol={} date={} page={} cursor={} retry={}/{} wait_ms={} error={}",
                            symbol, trade_date, page + 1, cursor, retry,
                            MAX_RATE_LIMIT_RETRIES, backoff_ms, error_message
                        );
                        tokio::time::sleep(std::time::Duration::from_millis(backoff_ms)).await;
                    } else {
                        return Err(std::io::Error::other(error_message).into());
                    }
                }
            }
        };

        let chunk_count = chunk.len();
        let response_first = chunk
            .first()
            .map(|c| format!("{} {}", c.stck_bsop_date, c.stck_cntg_hour))
            .unwrap_or_else(|| "none".to_string());
        let response_last = chunk
            .last()
            .map(|c| format!("{} {}", c.stck_bsop_date, c.stck_cntg_hour))
            .unwrap_or_else(|| "none".to_string());

        if chunk.is_empty() {
            tracing::warn!(
                target: "kis::night_candles",
                "[NIGHT-PAGE-EMPTY] symbol={} date={} page={} cursor={} elapsed_ms={} stop=empty_response",
                symbol, trade_date, page + 1, cursor, page_started.elapsed().as_millis()
            );
            break;
        }

        let mut earliest = String::new();
        let mut accepted = 0usize;
        let mut rejected_date = 0usize;
        let mut rejected_before_open = 0usize;
        for candle in &chunk {
            if candle.stck_bsop_date != trade_date {
                rejected_date += 1;
                continue;
            }
            if candle.stck_cntg_hour.as_str() < "180000" {
                rejected_before_open += 1;
                continue;
            }
            if earliest.is_empty() || candle.stck_cntg_hour < earliest {
                earliest = candle.stck_cntg_hour.clone();
            }
            result.push(to_candle(candle));
            accepted += 1;
        }

        tracing::info!(
            target: "kis::night_candles",
            "[NIGHT-PAGE-RESULT] symbol={} date={} page={} request_cursor={} response_count={} response_first={} response_last={} accepted={} rejected_date={} rejected_before_180000={} earliest_accepted={} accumulated={} elapsed_ms={}",
            symbol, trade_date, page + 1, cursor, chunk_count, response_first, response_last,
            accepted, rejected_date, rejected_before_open,
            if earliest.is_empty() { "none" } else { &earliest },
            result.len(), page_started.elapsed().as_millis()
        );

        if earliest.is_empty() {
            tracing::warn!(
                target: "kis::night_candles",
                "[NIGHT-PAGE-STOP] symbol={} date={} page={} reason=no_valid_candle_for_trade_date_and_session",
                symbol, trade_date, page + 1
            );
            break;
        }
        if earliest.as_str() <= "180000" {
            tracing::info!(
                target: "kis::night_candles",
                "[NIGHT-PAGE-STOP] symbol={} date={} page={} reason=session_open_reached earliest={}",
                symbol, trade_date, page + 1, earliest
            );
            break;
        }
        if page == 7 {
            tracing::warn!(
                target: "kis::night_candles",
                "[NIGHT-PAGE-STOP] symbol={} date={} page={} reason=max_pages_reached earliest={}",
                symbol, trade_date, page + 1, earliest
            );
            break;
        }

        let next_cursor = subtract_one_minute(&earliest);
        tracing::info!(
            target: "kis::night_candles",
            "[NIGHT-PAGE-NEXT] symbol={} date={} completed_page={} earliest={} next_cursor={} request_delay_ms={}",
            symbol, trade_date, page + 1, earliest, next_cursor, request_delay_ms
        );
        cursor = next_cursor;
    }

    result.sort_by(|a, b| a.time.cmp(&b.time));
    let before_dedup = result.len();
    result.dedup_by(|a, b| a.time == b.time);
    let first = result
        .first()
        .map(|c| c.time.clone())
        .unwrap_or_else(|| "none".to_string());
    let last = result
        .last()
        .map(|c| c.time.clone())
        .unwrap_or_else(|| "none".to_string());
    tracing::info!(
        target: "kis::night_candles",
        "[NIGHT-PAGINATION-DONE] symbol={} date={} raw_accumulated={} deduplicated={} duplicates_removed={} first_time={} last_time={}",
        symbol, trade_date, before_dedup, result.len(), before_dedup - result.len(), first, last
    );
    Ok(result)
}

/// 최근 실제 거래일 최대 2일과 장중 당일의 1분봉을 로드합니다.
pub async fn load_initial_candles(
    http_client: &reqwest::Client,
    app_key: &str,
    app_secret: &str,
    access_token: &str,
    symbols: &[FuturesSymbol],
) -> CandleStore {
    let mut candle_store = CandleStore::new();
    let today_dt = chrono::Utc::now().with_timezone(&chrono_tz::Asia::Seoul);
    let today = today_dt.format("%Y%m%d").to_string();
    let now_hhmm = today_dt
        .format("%H%M")
        .to_string()
        .parse::<u32>()
        .unwrap_or(0);
    let is_weekday = today_dt
        .format("%u")
        .to_string()
        .parse::<u32>()
        .unwrap_or(7)
        <= 5;
    let is_market_hours = is_weekday && (845..=1545).contains(&now_hhmm);

    for symbol in symbols {
        // 달력상 과거 14일에서 실제 분봉이 존재하는 서로 다른 거래일 2개를 찾습니다.
        let mut found: Vec<(String, Vec<kis::MinuteCandle>)> = Vec::new();
        for days_back in 1..=14 {
            let candidate_date = (today_dt.date_naive() - chrono::Duration::days(days_back))
                .format("%Y%m%d")
                .to_string();
            match kis::get_minute_chart(
                http_client,
                app_key,
                app_secret,
                access_token,
                &symbol.code,
                &candidate_date,
                "154500",
            )
            .await
            {
                Ok(chunk) => {
                    let latest_date = chunk
                        .iter()
                        .map(|c| c.stck_bsop_date.as_str())
                        .filter(|date| *date < today.as_str())
                        .max()
                        .map(str::to_string);
                    if let Some(date) = latest_date {
                        if !found.iter().any(|(existing, _)| existing == &date) {
                            let day_chunk: Vec<kis::MinuteCandle> = chunk
                                .into_iter()
                                .filter(|c| c.stck_bsop_date == date)
                                .collect();
                            if !day_chunk.is_empty() {
                                tracing::info!("거래일 탐색 확인: {} {}", symbol.code, date);
                                found.push((date, day_chunk));
                            }
                        }
                    }
                }
                Err(e) => {
                    tracing::warn!(
                        "거래일 탐색 중 1분봉 조회 실패({} / {}): {}",
                        symbol.code,
                        candidate_date,
                        e
                    );
                    // 휴장일 또는 특정 종목의 일시적인 API 오류일 수 있으므로
                    // 탐색을 중단하지 않고 다음 과거 날짜를 확인합니다.
                    tokio::time::sleep(std::time::Duration::from_millis(200)).await;
                    continue;
                }
            }
            if found.len() >= 2 {
                break;
            }
            tokio::time::sleep(std::time::Duration::from_millis(200)).await;
        }

        found.sort_by(|a, b| a.0.cmp(&b.0));
        if found.is_empty() {
            tracing::warn!(
                "직전 14일 내 거래일 1분봉을 찾지 못했습니다: {}",
                symbol.code
            );
        }
        for (trade_date, first_chunk) in found {
            tracing::info!("거래일({}) 1분봉 로드 시작: {}", trade_date, symbol.code);
            let candles = load_trade_date(
                http_client,
                app_key,
                app_secret,
                access_token,
                &symbol.code,
                &trade_date,
                first_chunk,
            )
            .await;
            tracing::info!("거래일 1분봉 로드 완료: {} {}건", trade_date, candles.len());
            candle_store.load_1m(&symbol.code, &trade_date, candles);
        }

        // 당일 장중 데이터는 최신 거래일로 저장되어 CandleStore에 함께 유지됩니다.
        if is_market_hours {
            tracing::info!("장중이므로 당일({}) 1분봉도 로드", today);
            let mut t_1m = Vec::new();
            let mut hour = format!("{:02}{:02}00", now_hhmm / 100, now_hhmm % 100);
            for _ in 0..5 {
                match kis::get_minute_chart(
                    http_client,
                    app_key,
                    app_secret,
                    access_token,
                    &symbol.code,
                    &today,
                    &hour,
                )
                .await
                {
                    Ok(chunk) => {
                        if chunk.is_empty() {
                            break;
                        }
                        let earliest = chunk
                            .iter()
                            .filter(|c| c.stck_bsop_date == today)
                            .map(|c| c.stck_cntg_hour.as_str())
                            .min()
                            .unwrap_or("")
                            .to_string();
                        for c in chunk.iter().filter(|c| c.stck_bsop_date == today) {
                            if c.stck_cntg_hour.len() < 4 {
                                continue;
                            }
                            t_1m.push(to_candle(c));
                        }
                        if earliest.is_empty() || earliest.as_str() <= "084500" {
                            break;
                        }
                        hour = subtract_one_minute(&earliest);
                    }
                    Err(e) => {

use super::{convert::to_candle, time::subtract_one_minute};
use crate::core::candles::Candle;
use crate::kis;
use reqwest::Client;

pub(super) async fn load_trade_date(
    http_client: &Client,
    app_key: &str,
    app_secret: &str,
    access_token: &str,
    symbol: &str,
    trade_date: &str,
    first_chunk: Vec<kis::MinuteCandle>,
) -> Vec<Candle> {
    let mut result = Vec::new();
    let mut first_chunk = Some(first_chunk);
    let mut hour = "154500".to_string();

    for page in 0..5 {
        let chunk = if page == 0 {
            first_chunk.take().unwrap_or_default()
        } else {
            match kis::get_minute_chart(
                http_client,
                app_key,
                app_secret,
                access_token,
                symbol,
                trade_date,
                &hour,
            )
            .await
            {
                Ok(chunk) => chunk,
                Err(e) => {
                    tracing::warn!("과거 1분봉 추가 조회 실패({} / {}): {}", symbol, hour, e);
                    break;
                }
            }
        };

        let mut earliest = String::new();
        for candle in chunk.iter().filter(|c| c.stck_bsop_date == trade_date) {
            if candle.stck_cntg_hour.len() < 6 {
                continue;
            }
            if earliest.is_empty() || candle.stck_cntg_hour < earliest {
                earliest = candle.stck_cntg_hour.clone();
            }
            result.push(to_candle(candle));
        }

        if earliest.is_empty() || earliest.as_str() <= "084500" {
            break;
        }
        hour = subtract_one_minute(&earliest);
        tokio::time::sleep(std::time::Duration::from_millis(200)).await;
    }

    result.sort_by(|a, b| a.time.cmp(&b.time));
    result.dedup_by(|a, b| a.time == b.time);
    result
}

/// "HHMMSS"에서 1분 전 계산
pub(super) fn subtract_one_minute(hhmmss: &str) -> String {
    if hhmmss.len() < 6 {
        return "154500".to_string();
    }
    let h: u32 = hhmmss[0..2].parse().unwrap_or(15);
    let m: u32 = hhmmss[2..4].parse().unwrap_or(45);
    let s: u32 = hhmmss[4..6].parse().unwrap_or(0);
    let total = h * 3600 + m * 60 + s;
    let new_total = if total >= 60 { total - 60 } else { 0 };
    format!(
        "{:02}{:02}{:02}",
        new_total / 3600,
        (new_total % 3600) / 60,
        new_total % 60
    )
}

use super::types::FuturesSymbol;

pub fn parse_master(text: &str) -> Vec<FuturesSymbol> {
    let mut result = Vec::new();
    for line in text.lines() {
        let fields: Vec<&str> = line.split('|').collect();
        if fields.len() < 4 {
            continue;
        }
        let info_type = fields[0].trim();
        if info_type != "1" && info_type != "B" {
            continue;
        }
        let code = fields[1].trim().to_string();
        let name = fields[3].trim().to_string();
        let month_code = name.split_whitespace().last().unwrap_or("");
        let month_ym = if month_code.len() == 6 {
            (
                month_code[0..4].parse::<i32>().unwrap_or(0),
                month_code[4..6].parse::<u32>().unwrap_or(0),
            )
        } else {
            (0, 0)
        };
        result.push(FuturesSymbol {
            info_type: info_type.to_string(),
            code,
            name,
            month_ym,
        });
    }
    result
}

pub fn pick_nearest(symbols: &[FuturesSymbol], info_type: &str) -> Option<FuturesSymbol> {
    let now = chrono::Local::now();
    let current_ym = (
        now.format("%Y").to_string().parse::<i32>().unwrap_or(0),
        now.format("%m").to_string().parse::<u32>().unwrap_or(0),
    );
    symbols
        .iter()
        .filter(|s| s.info_type == info_type && s.month_ym.0 > 0)
        .filter(|s| s.month_ym >= current_ym)
        .min_by_key(|s| s.month_ym)
        .cloned()
}

    };
    (status, Json(TicketError { code }))

use crate::chat::{ChatConnection, ChatError, ChatMessage, ChatService, validate_message_frame};
use axum::{
    Extension,
    extract::{
        Query,
        ws::{CloseFrame, Message, WebSocket, WebSocketUpgrade},
    },
    http::StatusCode,
    response::{IntoResponse, Response},
};
use serde::Deserialize;
use std::time::Duration;
use tokio::sync::broadcast;

const SEND_TIMEOUT: Duration = Duration::from_secs(5);

#[derive(Deserialize)]
pub(crate) struct TicketQuery {
    ticket: String,
}

pub(crate) async fn ws_handler(
    ws: WebSocketUpgrade,
    Query(query): Query<TicketQuery>,
    Extension(chat): Extension<ChatService>,
) -> Result<Response, StatusCode> {
    let connection = chat.consume_ticket(&query.ticket).map_err(ticket_status)?;

    Ok(ws
        .max_message_size(crate::chat::MAX_MESSAGE_FRAME_BYTES)
        .max_frame_size(crate::chat::MAX_MESSAGE_FRAME_BYTES)
        .on_upgrade(move |socket| handle_socket(socket, chat, connection))
        .into_response())
}

fn ticket_status(error: ChatError) -> StatusCode {
    match error {
        ChatError::InvalidTicket | ChatError::ExpiredTicket => StatusCode::UNAUTHORIZED,
        ChatError::RateLimited
        | ChatError::ConnectionLimitReached
        | ChatError::NicknameSequenceExhausted => StatusCode::TOO_MANY_REQUESTS,
        ChatError::InvalidClientAddress => StatusCode::BAD_REQUEST,
        _ => StatusCode::SERVICE_UNAVAILABLE,
    }
}

async fn handle_socket(mut socket: WebSocket, chat: ChatService, connection: ChatConnection) {
    let mut messages: broadcast::Receiver<ChatMessage> = chat.subscribe_messages();
    let mut presence = chat.subscribe_presence();
    while presence.try_recv().is_ok() {}
    if send_presence(&mut socket, chat.online_count())
        .await
        .is_err()
    {
        return;
    }
    loop {
        tokio::select! {
            incoming = socket.recv() => {
                match incoming {
                    Some(Ok(Message::Text(text))) => {
                        let text = match validate_message_frame(text.as_bytes()) {
                            Ok(text) => text,
                            Err(_) => {
                                close_with_policy_violation(&mut socket).await;
                                break;
                            }
                        };

                        if chat.publish_message(&connection, text).is_err() {
                            close_with_policy_violation(&mut socket).await;
                            break;
                        }
                    }
                    Some(Ok(Message::Close(_))) | None => break,
                    Some(Ok(Message::Ping(_))) | Some(Ok(Message::Pong(_))) => {}
                    Some(Ok(Message::Binary(_))) | Some(Err(_)) => {
                        close_with_policy_violation(&mut socket).await;
                        break;
                    }
                }
            }
            delivery = messages.recv() => {
                match delivery {
                    Ok(message) => {
                        if send_message(&mut socket, &message).await.is_err() {
                            break;
                        }
                    }
                    Err(broadcast::error::RecvError::Lagged(_)) => {
                        close_with_unavailable(&mut socket).await;
                        break;
                    }
                    Err(broadcast::error::RecvError::Closed) => break,
                }
            },
            count = presence.recv() => {
                match count {
                    Ok(online_count) => {
                        if send_presence(&mut socket, online_count).await.is_err() {
                            break;
                        }
                    }
                    Err(broadcast::error::RecvError::Lagged(_)) => {
                        if send_presence(&mut socket, chat.online_count()).await.is_err() {
                            break;
                        }
                    }
                    Err(broadcast::error::RecvError::Closed) => break,
                }
            }
        }
    }
}

async fn send_presence(socket: &mut WebSocket, online_count: usize) -> Result<(), ()> {
    let payload = serde_json::json!({
        "type": "presence",
        "online_count": online_count,
    });
    let payload = serde_json::to_string(&payload).map_err(|_| ())?;
    tokio::time::timeout(SEND_TIMEOUT, socket.send(Message::Text(payload.into())))
        .await
        .map_err(|_| ())?
        .map_err(|_| ())
}

async fn send_message(socket: &mut WebSocket, message: &ChatMessage) -> Result<(), ()> {
    let payload = serde_json::to_string(message).map_err(|_| ())?;
    tokio::time::timeout(SEND_TIMEOUT, socket.send(Message::Text(payload.into())))
        .await
        .map_err(|_| ())?
        .map_err(|_| ())
}

async fn close_with_policy_violation(socket: &mut WebSocket) {
    close_with(socket, 1008, "invalid chat frame").await;
}

async fn close_with_unavailable(socket: &mut WebSocket) {
    close_with(socket, 1013, "chat delivery unavailable").await;
}

async fn close_with(socket: &mut WebSocket, code: u16, reason: &'static str) {
    let frame = CloseFrame {
        code,
        reason: reason.into(),
    };
    let _ = tokio::time::timeout(SEND_TIMEOUT, socket.send(Message::Close(Some(frame)))).await;
}

pub async fn require_content_access(
    headers: &HeaderMap,
    state: &AppState,
) -> Result<(), (StatusCode, String)> {
    let claims = require_auth(headers, state).await?;
    let tier_query = format!("user_profiles?user_id=eq.{}&select=tier", claims.sub);
    let tier_resp = state
        .supabase
        .select(&tier_query)
        .await
        .map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, e))?;
    let tier = tier_resp
        .get(0)
        .and_then(|v| v.get("tier"))
        .and_then(|v| v.as_str())
        .unwrap_or("free");

    if tier == "free" {
        let kst_hour = chrono::Utc::now()
            .with_timezone(&chrono_tz::Asia::Seoul)
            .hour();
        if kst_hour < 14 {
            return Err((
                StatusCode::FORBIDDEN,
                "무료 회원은 오후 2시부터 열람 가능합니다".to_string(),
            ));
        }
    }
    Ok(())

/// UUID v4 형식의 글 ID인지 확인합니다.
pub fn is_uuid(value: &str) -> bool {
    value.len() == 36
        && value.bytes().enumerate().all(|(index, byte)| match index {
            8 | 13 | 18 | 23 => byte == b'-',
            _ => byte.is_ascii_hexdigit(),
        })
}

use crate::auth::jwt;
use crate::core::state::AppState;
use axum::http::HeaderMap;

/// Authorization 헤더에서 JWT 추출 + 검증
pub(super) async fn require_auth(
    headers: &HeaderMap,
    state: &AppState,
) -> Result<jwt::Claims, (axum::http::StatusCode, String)> {
    let auth = match headers.get("authorization").and_then(|v| v.to_str().ok()) {
        Some(value) => value,
        None => {
            tracing::warn!(target: "auth", "인증 실패: Authorization 헤더 누락 또는 유효하지 않은 헤더 값");
            return Err((
                axum::http::StatusCode::UNAUTHORIZED,
                "Authorization 헤더 필요".to_string(),
            ));
        }
    };

    let token = match auth.strip_prefix("Bearer ") {
        Some(value) if !value.is_empty() => value,
        _ => {
            tracing::warn!(target: "auth", "인증 실패: Bearer 토큰 형식 오류 또는 토큰 누락");
            return Err((
                axum::http::StatusCode::UNAUTHORIZED,
                "Bearer 토큰 필요".to_string(),
            ));
        }
    };

    jwt::verify_jwt(
        token,
        &state.jwt.legacy_hs256_secret,
        &state.supabase.url,
        &state.http_client,
    )
    .await
    .map_err(|error| {
        tracing::warn!(target: "auth", status = 401, error = %error, "요청 인증 거부");
        (axum::http::StatusCode::UNAUTHORIZED, error)
    })
}

    state: &AppState,
) -> Result<(), (axum::http::StatusCode, String)> {
    let query = format!("user_profiles?user_id=eq.{}&select=tier", claims.sub);
    let resp = state.supabase.select(&query).await.map_err(|error| {
        tracing::error!(target: "admin", error = %error, "관리자 권한 조회 중 Supabase 오류");

    })?;

    let tier = resp

        .and_then(|v| v.get("tier"))
        .and_then(|v| v.as_str())
        .unwrap_or("free");

    if tier != "admin" {
        tracing::warn!(target: "admin", tier = %tier, "관리자 API 접근 거부: user_profiles tier가 admin이 아님");
        return Err((
            axum::http::StatusCode::FORBIDDEN,
            "관리자 권한 필요".to_string(),

}

        ));
    }

    let query = if category.is_empty() {
        "notes?select=id,title,category,created_at&order=created_at.desc&limit=20".to_string()
    } else {
        format!(
            "notes?select=id,title,category,created_at&category=eq.{}&order=created_at.desc&limit=20",

    };

        .supabase

use serde::Deserialize;

const SUPPORTED_INTERVALS: &[&str] = &["1m", "2m", "5m", "10m", "30m", "60m"];
const SUPPORTED_DATES: &[&str] = &["yesterday", "today", "both"];

impl CandleParams {
    pub(crate) fn validate(&self) -> Result<(&str, &str, &str), &'static str> {
        let code = self
            .code
            .as_deref()
            .map(str::trim)
            .filter(|value| !value.is_empty())
            .ok_or("code 파라미터 필요")?;
        if code.len() > 32 || !code.bytes().all(|byte| byte.is_ascii_alphanumeric()) {
            return Err("올바른 code 파라미터가 필요합니다");
        }

        let interval = self.interval.as_deref().unwrap_or("1m");
        if !SUPPORTED_INTERVALS.contains(&interval) {
            return Err("지원하지 않는 interval입니다");
        }

        let date = self.date.as_deref().unwrap_or("both");
        if !SUPPORTED_DATES.contains(&date) {
            return Err("지원하지 않는 date 파라미터입니다");
        }

        Ok((code, interval, date))
    }
}

    store: &CandleStore,
    code: &str,
    interval: &str,
    date: &str,
    yesterday: &str,
    today: &str,
) -> Vec<Value> {
    match date {
        "yesterday" => to_json(&store.get(code, yesterday, interval), yesterday),
        "today" => to_json(&store.get(code, today, interval), today),
        _ => {
            // Return the latest dates actually present in CandleStore, independent of weekdays.
            let mut all = Vec::new();
            for stored_date in store.dates(code) {
                all.extend(to_json(
                    &store.get(code, &stored_date, interval),
                    &stored_date,
                ));
            }
            all
        }
    }
}

//! KIS 1분봉 과거 데이터 검증을 백그라운드에서 실행하고 로그 파일에 출력합니다.

use chrono::NaiveTime;
use crate::{core::{candles::Candle, state::AppState}, kis};
use axum::{Json, extract::State, http::StatusCode};
use serde_json::{Value, json};
use std::{
    collections::{BTreeMap, BTreeSet},

        Arc,
        atomic::Ordering,
    },

const DAY_START: &str = "084500";
const MAX_PAGES: usize = 8;
const MAX_RATE_LIMIT_RETRIES: usize = 3;

struct RunningGuard(Arc<std::sync::atomic::AtomicBool>);

impl Drop for RunningGuard {
    fn drop(&mut self) {
        self.0.store(false, Ordering::Release);
    }
}

/// POST /api/admin/market/minute-candle-check
/// 응답은 즉시 반환하고, 실제 REST 조회와 로그 출력은 독립 Tokio task에서 수행합니다.
pub async fn start_minute_candle_check(
    State(state): State<AppState>,
) -> (StatusCode, Json<Value>) {
    if state
        .minute_candle_check_running
        .compare_exchange(false, true, Ordering::AcqRel, Ordering::Acquire)
        .is_err()
    {
        return (
            StatusCode::CONFLICT,
            Json(json!({
                "ok": false,
                "running": true,
                "message": "분봉 검증 작업이 이미 실행 중입니다."
            })),
        );
    }

    let running = state.minute_candle_check_running.clone();
    tokio::spawn(async move {
        let _guard = RunningGuard(running);
        if let Err(error) = run_minute_candle_check(state).await {
            tracing::error!(target: "minute_candle_validation", "[MINUTE-CHECK] 검증 작업 실패: {}", error);
        }
    });

    (
        StatusCode::ACCEPTED,
        Json(json!({
            "ok": true,
            "running": true,
            "message": "분봉 검증을 백그라운드에서 시작했습니다. backend 로그 파일을 확인하세요."
        })),
    )
}

async fn run_minute_candle_check(state: AppState) -> Result<(), String> {
    let now = chrono::Utc::now().with_timezone(&chrono_tz::Asia::Seoul);
    let start = NaiveTime::parse_from_str(DAY_START, "%H%M%S")
        .map_err(|error| error.to_string())?;
    if now.time() < start {
        return Err("현재 시각이 08:45 이전이므로 오늘 주간 분봉을 조회하지 않습니다.".to_string());
    }

    let date = now.format("%Y%m%d").to_string();
    let end_time = now.format("%H%M%S").to_string();

    tracing::info!(
        target: "minute_candle_validation",
        "========== KIS 1분봉 검증 시작: date={}, range={}~{} KST ==========",
        date, DAY_START, end_time
    );

    let symbols = [&state.regular, &state.mini];
    for symbol in symbols {
        tracing::info!(
            target: "minute_candle_validation",
            "----- 종목 시작: {} ({}) -----",
            symbol.name, symbol.code
        );

        match fetch_day_candles(
            &state,
            &symbol.code,
            &date,
            &end_time,
        )
        .await
        {
            Ok(candles) if candles.is_empty() => {
                tracing::info!(
                    target: "minute_candle_validation",
                    "[MINUTE-CHECK] {} ({}) : 08:45~현재 조회 데이터 없음. 장 시작 지연 또는 미수신 가능성이 있으며 오류로 판정하지 않습니다.",
                    symbol.name, symbol.code
                );
            }
            Ok(candles) => {
                // 내부 CandleStore는 짧게 잠가 스냅샷만 복사하고, 이후 비교/로그 중에는 잠금을 유지하지 않습니다.
                let local_candles = {
                    let store = state.candles.lock().await;
                    store.get(&symbol.code, &date, "1m")
                };
                log_and_compare_candles(
                    &symbol.name,
                    &symbol.code,
                    &candles,
                    &local_candles,
                    &end_time,
                );
            }
            Err(error) => {
                tracing::error!(
                    target: "minute_candle_validation",
                    "[MINUTE-CHECK] {} ({}) 조회 실패: {}",
                    symbol.name, symbol.code, error
                );
            }
        }
    }

    tracing::info!(
        target: "minute_candle_validation",
        "========== KIS 1분봉 검증 종료: date={} ==========",
        date
    );
    Ok(())
}

fn log_and_compare_candles(
    name: &str,
    symbol: &str,
    broker: &BTreeMap<String, kis::MinuteCandle>,
    local: &[Candle],
    end_time: &str,
) {
    let local_map: BTreeMap<String, &Candle> = local
        .iter()
        .filter(|candle| candle.time.len() == 4)
        .map(|candle| (format!("{}00", candle.time), candle))
        .filter(|(time, _)| time.as_str() >= DAY_START && time.as_str() <= end_time)
        .collect();

    let mut times = BTreeSet::new();
    times.extend(broker.keys().cloned());
    times.extend(local_map.keys().cloned());

    let first_broker = broker.keys().next().cloned().unwrap_or_else(|| "없음".to_string());
    let first_local = local_map.keys().next().cloned().unwrap_or_else(|| "없음".to_string());
    tracing::info!(
        target: "minute_candle_validation",
        "[MINUTE-CHECK] {} ({}) : KIS 최초={}, 내부 최초={}, KIS {}개 / 내부 {}개",
        name, symbol, first_broker, first_local, broker.len(), local_map.len()
    );
    tracing::info!(
        target: "minute_candle_validation",
        "시간 | KIS 시가/고가/저가/종가 | 내부 시가/고가/저가/종가 | 판정"
    );

    let current_minute = &end_time[..4];
    let mut matched = 0usize;
    let mut mismatched = 0usize;
    let mut broker_only = 0usize;
    let mut local_only = 0usize;
    let mut live = 0usize;
    let mut log_batch = Vec::with_capacity(25);

    for time in times {
        let broker_candle = broker.get(&time);
        let local_candle = local_map.get(&time);
        let time_label = &time[..4];

        let (broker_values, local_values, status) = match (broker_candle, local_candle) {
            (Some(reference), Some(ours)) if time_label == current_minute => {
                live += 1;
                (
                    format!("{}/{}/{}/{}", reference.futs_oprc, reference.futs_hgpr, reference.futs_lwpr, reference.futs_prpr),
                    format!("{:.2}/{:.2}/{:.2}/{:.2}", ours.open, ours.high, ours.low, ours.close),
                    "진행중",
                )
            }
            (Some(reference), Some(ours)) => {
                let same = [
                    (reference.futs_oprc.parse::<f64>(), ours.open),
                    (reference.futs_hgpr.parse::<f64>(), ours.high),
                    (reference.futs_lwpr.parse::<f64>(), ours.low),
                    (reference.futs_prpr.parse::<f64>(), ours.close),
                ]
                .iter()
                .all(|(reference, ours)| {
                    reference.as_ref().map(|value| (*value - *ours).abs() < 0.000001).unwrap_or(false)
                });
                if same {
                    matched += 1;
                    (
                        format!("{}/{}/{}/{}", reference.futs_oprc, reference.futs_hgpr, reference.futs_lwpr, reference.futs_prpr),
                        format!("{:.2}/{:.2}/{:.2}/{:.2}", ours.open, ours.high, ours.low, ours.close),
                        "일치",
                    )
                } else {
                    mismatched += 1;
                    (
                        format!("{}/{}/{}/{}", reference.futs_oprc, reference.futs_hgpr, reference.futs_lwpr, reference.futs_prpr),
                        format!("{:.2}/{:.2}/{:.2}/{:.2}", ours.open, ours.high, ours.low, ours.close),
                        "불일치",
                    )
                }
            }
            (Some(reference), None) => {
                broker_only += 1;
                (
                    format!("{}/{}/{}/{}", reference.futs_oprc, reference.futs_hgpr, reference.futs_lwpr, reference.futs_prpr),
                    "-".to_string(),
                    "KIS만 있음",
                )
            }
            (None, Some(ours)) => {
                local_only += 1;
                (
                    "-".to_string(),
                    format!("{:.2}/{:.2}/{:.2}/{:.2}", ours.open, ours.high, ours.low, ours.close),
                    "내부만 있음",
                )
            }
            (None, None) => continue,
        };

        log_batch.push(format!(
            "{} | {} | {} | {}",
            time, broker_values, local_values, status
        ));

        // 비동기 파일 로거의 큐에 수백 개 이벤트를 한꺼번에 밀어 넣지 않도록
        // 25개 분봉씩 묶어 출력합니다. 실시간 WebSocket 처리와도 별도 task입니다.
        if log_batch.len() == 25 {
            tracing::info!(
                target: "minute_candle_validation",
                "[MINUTE-CHECK-ROWS] {} ({})\n{}",
                name, symbol, log_batch.join("\n")
            );
            log_batch.clear();
        }
    }

    if !log_batch.is_empty() {
        tracing::info!(
            target: "minute_candle_validation",
            "[MINUTE-CHECK-ROWS] {} ({})\n{}",
            name, symbol, log_batch.join("\n")
        );
    }

    tracing::info!(
        target: "minute_candle_validation",
        "[MINUTE-CHECK-SUMMARY] {} ({}) 일치={} 불일치={} KIS만={} 내부만={} 진행중={}",
        name, symbol, matched, mismatched, broker_only, local_only, live
    );
}

async fn fetch_day_candles(
    state: &AppState,
    symbol: &str,
    date: &str,
    end_time: &str,
) -> Result<BTreeMap<String, kis::MinuteCandle>, String> {
    let mut result = BTreeMap::new();
    let mut cursor = end_time.to_string();

    for page in 0..MAX_PAGES {
        let request_delay_ms = std::env::var("KIS_CANDLE_REQUEST_DELAY_MS")
            .ok()
            .and_then(|value| value.parse::<u64>().ok())
            .unwrap_or(300);

        let mut retry = 0usize;
        let candles = loop {
            match kis::get_minute_chart_for_market(
                &state.http_client,
                &state.kis.app_key,
                &state.kis.app_secret,
                &state.kis.access_token,
                "F",
                symbol,
                date,
                &cursor,
            )
            .await
            {
                Ok(candles) => break candles,
                Err(error) => {
                    let message = error.to_string();
                    drop(error);
                    if message.contains("EGW00201") && retry < MAX_RATE_LIMIT_RETRIES {
                        retry += 1;
                        let backoff_ms = request_delay_ms
                            .max(1000u64.saturating_mul(2u64.pow(retry as u32 - 1)));
                        tracing::warn!(
                            target: "minute_candle_validation",
                            "[MINUTE-CHECK-RATE-LIMIT] symbol={} date={} cursor={} retry={}/{} wait_ms={}",
                            symbol, date, cursor, retry, MAX_RATE_LIMIT_RETRIES, backoff_ms
                        );
                        tokio::time::sleep(std::time::Duration::from_millis(backoff_ms)).await;
                    } else {
                        return Err(message);
                    }
                }
            }
        };

        tokio::time::sleep(std::time::Duration::from_millis(request_delay_ms)).await;

        if candles.is_empty() {
            break;
        }

        let earliest = candles
            .iter()
            .filter(|candle| candle.stck_bsop_date == date)
            .map(|candle| candle.stck_cntg_hour.as_str())
            .min()
            .map(str::to_string);

        for candle in candles {
            if candle.stck_bsop_date == date
                && candle.stck_cntg_hour.as_str() >= DAY_START
                && candle.stck_cntg_hour.as_str() <= end_time
            {
                result.insert(candle.stck_cntg_hour.clone(), candle);
            }
        }

        let Some(earliest) = earliest else {
            break;
        };

        if earliest.as_str() <= DAY_START {
            break;
        }


        let next_cursor = previous_minute(&earliest)
            .ok_or_else(|| format!("잘못된 분봉 시간 형식: {}", earliest))?;
        if next_cursor >= cursor {
            tracing::warn!(
                target: "minute_candle_validation",
                "[MINUTE-CHECK] 페이지 커서 진행 중단: symbol={}, cursor={}, next={}",
                symbol, cursor, next_cursor
            );
            break;
        }

        cursor = next_cursor;
        if page + 1 == MAX_PAGES {
            tracing::warn!(
                target: "minute_candle_validation",
                "[MINUTE-CHECK] 최대 페이지 수 도달: symbol={}, max_pages={}, earliest_seen={}",
                symbol, MAX_PAGES, earliest
            );
        }
    }

    if !result.is_empty() {
        tracing::info!(
            target: "minute_candle_validation",
            "[MINUTE-CHECK] symbol={} 누적 조회 {}개 분봉",
            symbol, result.len()
        );
    }
    Ok(result)
}

fn previous_minute(value: &str) -> Option<String> {
    if value.len() != 6 || !value.bytes().all(|byte| byte.is_ascii_digit()) {
        return None;
    }
    let hour = value[0..2].parse::<u32>().ok()?;
    let minute = value[2..4].parse::<u32>().ok()?;
    let second = value[4..6].parse::<u32>().ok()?;
    if hour > 23 || minute > 59 || second > 59 {
        return None;
    }
    let total_seconds = hour * 3600 + minute * 60 + second;
    let previous = total_seconds.checked_sub(60)?;
    Some(format!(
        "{:02}{:02}{:02}",
        previous / 3600,
        (previous % 3600) / 60,
        previous % 60
    ))
}

    symbol: &FuturesSymbol,

pub(super) async fn build_prices(state: &AppState) -> Value {
    let (regular_pair, mini_pair) = tokio::join!(
        fetch_pair(state, &state.regular),
        fetch_pair(state, &state.mini),
    );
    let (regular_day, regular_night) = regular_pair;
    let (mini_day, mini_night) = mini_pair;

    let regular_reference = regular_day.as_ref().ok().map(|p| p.price.clone());
    let mini_reference = mini_day.as_ref().ok().map(|p| p.price.clone());

    let regular_night_json = night_result(regular_night, regular_reference);
    let mini_night_json = night_result(mini_night, mini_reference);

    json!({
        "regular": match regular_day {
            Ok(price) => json!({ "ok": true, "data": price, "night": regular_night_json }),
            Err(error) => json!({ "ok": false, "error": error, "night": regular_night_json }),
        },
        "mini": match mini_day {
            Ok(price) => json!({ "ok": true, "data": price, "night": mini_night_json }),
            Err(error) => json!({ "ok": false, "error": error, "night": mini_night_json }),
        },
    })
}

use super::{prices::build_prices, ws_message::tick_message};
use crate::core::state::AppState;
use axum::{
    extract::{
        State,
        ws::{Message, WebSocket, WebSocketUpgrade},
    },
    response::IntoResponse,
};
use serde_json::json;

pub async fn ws_handler(ws: WebSocketUpgrade, State(state): State<AppState>) -> impl IntoResponse {
    ws.on_upgrade(move |socket| handle_socket(socket, state))
}

async fn handle_socket(mut socket: WebSocket, state: AppState) {
    tracing::info!("브라우저 WebSocket 연결됨");
    // 먼저 구독해 스냅샷 조회 중 발생한 체결도 버퍼에서 이어받습니다.
    let mut rx = state.ws_tx.subscribe();

    let snapshot = build_prices(&state).await;
    let range_snapshot = state.session_range.lock().await.snapshot();
    let snapshot_message = serde_json::to_string(&json!({
        "type": "snapshot",
        "data": snapshot,
        "range": range_snapshot,
    }))
    .unwrap_or_default();

    if socket
        .send(Message::Text(snapshot_message.into()))
        .await
        .is_err()
    {
        tracing::warn!("브라우저 WebSocket 초기 시세 스냅샷 전송 실패");
        return;
    }

    loop {
        tokio::select! {
            result = rx.recv() => {
                match result {
                    Ok(tick) => {
                        let range_snapshot = state.session_range.lock().await.snapshot();
                        let message = tick_message(&tick, &range_snapshot);
                        if socket.send(Message::Text(message.into())).await.is_err() { break; }
                    }
                    Err(tokio::sync::broadcast::error::RecvError::Lagged(skipped)) => {
                        // 느린 브라우저 연결은 누락된 틱만 건너뛰고 다음 최신 틱 수신을 계속합니다.
                        tracing::warn!("브라우저 WebSocket 수신 지연: {}개 틱 누락", skipped);
                        continue;
                    }
                    Err(tokio::sync::broadcast::error::RecvError::Closed) => break,
                }
            }
            msg = socket.recv() => {
                match msg {
                    Some(Ok(Message::Close(_))) | None => break,
                    _ => {}
                }
            }
        }
    }
    tracing::info!("브라우저 WebSocket 연결 종료");
}

// 설명: Axum 서버 실행 및 CORS 설정
//       - TCP Listener를 초기화보다 먼저 열어 연결 거부 방지

use axum::http::{HeaderValue, Method, header};
use std::future::{Future, IntoFuture};
use std::net::SocketAddr;
use tower_http::cors::CorsLayer;

pub async fn serve_with_startup<F>(initialize: F) -> Result<(), Box<dyn std::error::Error>>
where
    F: Future<Output = Result<crate::core::state::AppState, Box<dyn std::error::Error>>>,
{
    let cors_origins = std::env::var("CORS_ALLOWED_ORIGINS")
        .unwrap_or_else(|_| "http://localhost:5173,http://127.0.0.1:5173".to_string());
    let allowed_origins = cors_origins
        .split(',')
        .map(str::trim)
        .filter(|origin| !origin.is_empty())
        .map(|origin| {
            origin.parse::<HeaderValue>().map_err(|e| {
                tracing::error!("잘못된 CORS Origin 설정({}): {}", origin, e);
                std::io::Error::new(
                    std::io::ErrorKind::InvalidInput,
                    format!(
                        "CORS_ALLOWED_ORIGINS에 잘못된 Origin이 있습니다: {}",
                        origin
                    ),
                )
            })
        })
        .collect::<Result<Vec<_>, _>>()?;

    if allowed_origins.is_empty() {
        let message = "CORS_ALLOWED_ORIGINS에 허용 Origin을 하나 이상 설정해야 합니다.";
        tracing::error!("{}", message);
        return Err(std::io::Error::new(std::io::ErrorKind::InvalidInput, message).into());
    }

    let cors = CorsLayer::new()
        .allow_origin(allowed_origins)
        .allow_methods([
            Method::GET,
            Method::POST,
            Method::PUT,
            Method::DELETE,
            Method::OPTIONS,
        ])
        .allow_headers([header::AUTHORIZATION, header::CONTENT_TYPE]);

    let bind_addr =
        std::env::var("BACKEND_BIND_ADDR").unwrap_or_else(|_| "127.0.0.1:8080".to_string());
    let listener = tokio::net::TcpListener::bind(&bind_addr)
        .await
        .map_err(|e| {
            tracing::error!("백엔드 서버 주소({}) 바인딩 실패: {}", bind_addr, e);
            e
        })?;

    let startup = super::startup::StartupShared::new();
    let app = super::startup::build(startup.clone()).layer(cors);

    tracing::info!(
        "backend listening on {} (startup health endpoint active)",
        bind_addr
    );

    // Axum 0.8의 Serve는 Future가 아니라 IntoFuture를 구현하므로
    // select!에서 사용할 Future로 변환합니다.
    let server = axum::serve(
        listener,
        app.into_make_service_with_connect_info::<SocketAddr>(),
    )
    .into_future();
    tokio::pin!(server);
    tokio::pin!(initialize);

    // HTTP 서버와 외부 서비스 초기화를 같은 Tokio task에서 동시에 진행합니다.
    // initialize Future를 별도 tokio::spawn에 넘기지 않아도 되므로
    // 기존 Box<dyn Error> 반환 타입을 유지할 수 있습니다.
    tokio::select! {
        result = &mut server => {
            result.map_err(|e| {
                tracing::error!("Axum 서버 실행 중 오류: {}", e);
                e
            })?;
            return Ok(());
        }
        result = &mut initialize => {
            match result {
                Ok(state) => {
                    let regular = state.regular.name.clone();
                    let mini = state.mini.name.clone();
                    let app = super::router::build(state);
                    startup.mark_ready(app, regular, mini).await;
                    tracing::info!("Backend 초기화 완료 - API 요청 처리 준비 완료");
                }
                Err(error) => {
                    tracing::error!("Backend 초기화 실패: {}", error);
                    startup.mark_failed().await;
                    tracing::error!(
                        "Health Check는 유지됩니다. 원인 해결 후 Backend를 재시작해 주세요."
                    );
                }
            }
        }
    }

    server.await.map_err(|e| {
        tracing::error!("Axum 서버 실행 중 오류: {}", e);
        e
    })?;

    Ok(())
}

// ============================================================
// 파일명: server/startup.rs
// 설명: 초기화 중에도 Health Check를 제공하는 서버 진입 라우터
//       - HTTP Listener를 먼저 열어 ECONNREFUSED 방지
//       - 초기화 완료 전 API 요청은 503 + Retry-After 반환
//       - 준비 완료 후 요청을 실제 애플리케이션 Router로 전달

use axum::{
    Json, Router,
    body::Body,
    extract::State,
    http::{Request, StatusCode, header},
    response::{IntoResponse, Response},
    routing::get,
};
use serde_json::json;
use std::sync::Arc;
use tokio::sync::RwLock;
use tower::ServiceExt;

#[derive(Clone)]
pub struct StartupShared {
    app: Arc<RwLock<Option<Router>>>,
    status: Arc<RwLock<StartupStatus>>,
}

#[derive(Clone)]
enum StartupStatus {
    Starting,
    Ready { regular: String, mini: String },
    Failed,
}

impl StartupShared {
    pub fn new() -> Self {
        Self {
            app: Arc::new(RwLock::new(None)),
            status: Arc::new(RwLock::new(StartupStatus::Starting)),
        }
    }

    pub async fn mark_ready(&self, app: Router, regular: String, mini: String) {
        *self.app.write().await = Some(app);
        *self.status.write().await = StartupStatus::Ready { regular, mini };
    }

    pub async fn mark_failed(&self) {
        *self.status.write().await = StartupStatus::Failed;
    }
}

async fn health(State(shared): State<StartupShared>) -> Response {
    let status = shared.status.read().await.clone();
    match status {
        StartupStatus::Starting => (
            StatusCode::SERVICE_UNAVAILABLE,
            [(header::RETRY_AFTER, "2")],
            Json(json!({
                "status": "starting",
                "ready": false,
                "message": "Backend 초기화가 진행 중입니다."
            })),
        )
            .into_response(),
        StartupStatus::Ready { regular, mini } => (
            StatusCode::OK,
            Json(json!({
                "status": "ok",
                "ready": true,
                "regular": regular,
                "mini": mini
            })),
        )
            .into_response(),
        StartupStatus::Failed => (
            StatusCode::SERVICE_UNAVAILABLE,
            [(header::RETRY_AFTER, "5")],
            Json(json!({
                "status": "failed",
                "ready": false,
                "message": "Backend 초기화에 실패했습니다. 백엔드 로그를 확인해 주세요."
            })),
        )
            .into_response(),
    }
}

async fn dispatch(State(shared): State<StartupShared>, request: Request<Body>) -> Response {
    let app = shared.app.read().await.clone();

    if let Some(app) = app {
        return match app.oneshot(request).await {
            Ok(response) => response,
            Err(never) => match never {},
        };
    }

    let status = shared.status.read().await.clone();
    let message = match status {
        StartupStatus::Starting => "Backend 초기화가 진행 중입니다.",
        StartupStatus::Ready { .. } => "Backend 서비스가 준비 중입니다.",
        StartupStatus::Failed => "Backend 초기화에 실패했습니다. 백엔드 로그를 확인해 주세요.",
    };

    (
        StatusCode::SERVICE_UNAVAILABLE,
        [(header::RETRY_AFTER, "2")],
        Json(json!({
            "status": "unavailable",
            "ready": false,
            "message": message
        })),
    )
        .into_response()
}

pub fn build(shared: StartupShared) -> Router {
    Router::new()
        .route("/api/health", get(health))
        .fallback(dispatch)
        .with_state(shared)
}
