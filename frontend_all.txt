===== frontend\vite.config.ts =====


import { defineConfig, loadEnv } from 'vite'
import react from '@vitejs/plugin-react'

export default defineConfig(({ mode }) => {
  const env = loadEnv(mode, process.cwd(), '')
  const backendTarget = env.VITE_DEV_BACKEND_URL || 'http://127.0.0.1:8080'
  const frontendPort = Number(env.VITE_PORT || 5173)

  return {
    plugins: [react()],
    server: {
      port: frontendPort,
      proxy: {
        '/api': {
          target: backendTarget,
          changeOrigin: true,
        },
        '/ws': {
          target: backendTarget.replace(/^http/, 'ws'),
          ws: true,
        },
      },
    },
  }
})

//       - 시세/채팅 패널은 항상 마운트하여 화면 전환 중에도 상태와 연결 유지
//       - 오른쪽 콘텐츠 영역만 전환

import { useEffect, useRef, useState } from 'react';
import { useAuthSession } from './features/auth/hooks/useAuthSession';
import { useAdminRole } from './features/auth/hooks/useAdminRole';
import PricesTab from './features/prices/PricesTab';
import AppHeader from './components/AppHeader';
import AppModal, { type ActiveModal } from './components/AppModal';
import './App.css';
import { useBackendHealth } from './hooks/useBackendHealth';

export default function App() {
  const { session, authLoading } = useAuthSession();
  const backendHealth = useBackendHealth();
  const backendReady = backendHealth === 'ready';
  const [activePage, setActivePage] = useState<'prices' | 'chat'>('chat');
  const [activeModal, setActiveModal] = useState<ActiveModal>(null);
  const [isMarketConnected, setMarketConnected] = useState(false);
  const previousSessionRef = useRef(session);
  const isAdmin = useAdminRole(session);
  const selectPage = (page: 'prices' | 'chat') => {
    setActivePage(page);
    setActiveModal(null);
  };

  // Google OAuth 팝업에서 인증이 완료되면 이 창의 세션도 공유되고, 팝업은 자동으로 닫습니다.
  useEffect(() => {
    if (window.opener && session) {
      window.opener.postMessage({ type: 'supabase-auth-complete' }, window.location.origin);
      window.close();
    }
  }, [session]);

  // 비회원 상태에서 로그인에 성공하면 로그인 유도 모달만 닫고 시세 화면에 머뭅니다.
  // 분석 화면은 사용자가 분석 버튼을 다시 눌렀을 때만 엽니다.
  useEffect(() => {
    const justSignedIn = !previousSessionRef.current && !!session;
    previousSessionRef.current = session;

    if (justSignedIn && (activeModal === 'login' || activeModal === 'analysis')) {
      setActiveModal(null);
    }
  }, [session, activeModal]);

  useEffect(() => {
    if (!activeModal) return;

    const handleKeyDown = (event: KeyboardEvent) => {
      if (event.key === 'Escape') setActiveModal(null);
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [activeModal]);

  // OAuth 팝업은 인증 세션을 확인한 뒤 부모 창으로 결과를 전달하고 닫힙니다.
  if (!authLoading && window.opener && session) {
    return <p style={{ padding: 20, color: '#e2e8f0' }}>로그인 완료. 창을 닫는 중...</p>;
  }

  return (
    <div className="app">
      <AppHeader
        activeModal={activeModal}
        isAdmin={isAdmin}
        session={session}
        isMarketConnected={isMarketConnected}
        authLoading={authLoading}
        activePage={activePage}
        onSelectPage={selectPage}
        onOpenModal={setActiveModal}
      />

      {/* PricesTab 내부에서 왼쪽 시세와 오른쪽 두 패널을 함께 유지합니다. */}
      <PricesTab activePage={activePage} backendReady={backendReady} onMarketConnectionChange={setMarketConnected} />

      {!authLoading && (
        <AppModal
          activeModal={activeModal}
          session={session}
          isAdmin={isAdmin}
          onClose={() => setActiveModal(null)}
        />
      )}
    </div>
  );
}

// ============================================================
// 파일명: CandleChart.tsx
// 설명: TradingView Lightweight Charts 캔들 차트
//       - 배경 검정, 양봉 초록, 음봉 빨강
//       - 종목/interval 변경 시 차트 재생성
//       - 컨테이너 크기 변화에 반응 (ResizeObserver)
//       - date(YYYYMMDD) + time(HHMMSS) → UTC timestamp
// 작성일: 2026-11-14
// ============================================================

import { useEffect, useRef } from 'react';
import {
  createChart,
  CandlestickSeries,
  LineStyle,
  type ISeriesApi,
  type UTCTimestamp,
} from 'lightweight-charts';
import { candleColors } from './candleColors';

interface Candle {
  date: string;   // "YYYYMMDD" (예: "20261113")
  time: string;   // "HHMMSS" (예: "093000")
  open: number;
  high: number;
  low: number;
  close: number;
  volume: number;
}

interface RangeLine {
  label: string;
  price: number;
  color: string;
}

interface Props {
  data: Candle[];
  interval: string;
  symbolKey: string;
  rangeLines: RangeLine[];
}

function toChartData(data: Candle[]) {
  return data.map((c) => {
    const y = Number(c.date.slice(0, 4));
    const m = Number(c.date.slice(4, 6));
    const d = Number(c.date.slice(6, 8));
    const hh = Number(c.time.slice(0, 2));
    const mm = Number(c.time.slice(2, 4));
    const ts = Math.floor(Date.UTC(y, m - 1, d, hh, mm) / 1000);
    return {
      time: ts as UTCTimestamp,
      open: c.open,
      high: c.high,
      low: c.low,
      close: c.close,
    };
  });
}

function sameCandle(
  left: ReturnType<typeof toChartData>[number],
  right: ReturnType<typeof toChartData>[number],
): boolean {
  return left.time === right.time
    && left.open === right.open
    && left.high === right.high
    && left.low === right.low
    && left.close === right.close;
}

export default function CandleChart({ data, interval, symbolKey, rangeLines }: Props) {
  const containerRef = useRef<HTMLDivElement>(null);
  const chartRef = useRef<ReturnType<typeof createChart> | null>(null);
  const candleSeriesRef = useRef<ISeriesApi<'Candlestick'> | null>(null);
  const priceLinesRef = useRef<ReturnType<ISeriesApi<'Candlestick'>['createPriceLine']>[]>([]);
  const chartDataRef = useRef<ReturnType<typeof toChartData> | null>(null);
  const hasFittedVisibleContentRef = useRef(false);

  useEffect(() => {
    if (!containerRef.current) return;

    const chart = createChart(containerRef.current, {
      width: containerRef.current.clientWidth,
      height: containerRef.current.clientHeight,
      layout: {
        background: { color: '#000000' },
        textColor: '#e2e8f0',
      },
      grid: {
        vertLines: { color: '#1e293b' },
        horzLines: { color: '#1e293b' },
      },
      timeScale: {
        timeVisible: true,
        secondsVisible: false,
        shiftVisibleRangeOnNewBar: false,
      },
    });

    const candleSeries = chart.addSeries(CandlestickSeries, {
      upColor: candleColors.up,
      downColor: candleColors.down,
      borderVisible: false,
      wickUpColor: candleColors.up,
      wickDownColor: candleColors.down,
    });

    chartRef.current = chart;
    candleSeriesRef.current = candleSeries;
    chartDataRef.current = null;
    hasFittedVisibleContentRef.current = false;

    const resizeObserver = new ResizeObserver((entries) => {
      for (const entry of entries) {
        const { width, height } = entry.contentRect;
        if (width > 0 && height > 0) {
          chart.applyOptions({ width, height });
          // The chart may be mounted inside a hidden panel. Fit again when it first becomes visible.
          if (!hasFittedVisibleContentRef.current && chartDataRef.current?.length) {
            chart.timeScale().fitContent();
            hasFittedVisibleContentRef.current = true;
          }
        }
      }
    });
    resizeObserver.observe(containerRef.current);

    return () => {
      resizeObserver.disconnect();
      chart.remove();
      chartRef.current = null;
      candleSeriesRef.current = null;
      priceLinesRef.current = [];
      chartDataRef.current = null;
    };
  }, [interval, symbolKey]);

  useEffect(() => {
    const chart = chartRef.current;
    const candleSeries = candleSeriesRef.current;
    if (!chart || !candleSeries || data.length === 0) return;

    const nextData = toChartData(data);
    const previousData = chartDataRef.current;
    if (!previousData) {
      candleSeries.setData(nextData);
      if (containerRef.current && containerRef.current.clientWidth > 0) {
        chart.timeScale().fitContent();
        hasFittedVisibleContentRef.current = true;
      }
      chartDataRef.current = nextData;
      return;
    }

    const commonLength = Math.min(previousData.length, nextData.length);
    let unchangedPrefixLength = 0;
    while (
      unchangedPrefixLength < commonLength
      && sameCandle(previousData[unchangedPrefixLength], nextData[unchangedPrefixLength])
    ) {
      unchangedPrefixLength += 1;
    }

    if (
      unchangedPrefixLength === previousData.length
      || (
        previousData.length === nextData.length
        && unchangedPrefixLength === previousData.length - 1
      )
    ) {
      for (let index = unchangedPrefixLength; index < nextData.length; index += 1) {
        candleSeries.update(nextData[index]);
      }
    } else {
      const visibleRange = chart.timeScale().getVisibleLogicalRange();
      candleSeries.setData(nextData);
      if (visibleRange) {
        chart.timeScale().setVisibleLogicalRange(visibleRange);
      }
    }

    chartDataRef.current = nextData;
  }, [data]);

  useEffect(() => {
    const candleSeries = candleSeriesRef.current;
    if (!candleSeries) return;

    for (const line of priceLinesRef.current) {
      candleSeries.removePriceLine(line);
    }
    priceLinesRef.current = rangeLines.map((line) =>
      candleSeries.createPriceLine({
        price: line.price,
        color: line.color,
        lineWidth: 1,
        lineStyle: LineStyle.Dashed,
        axisLabelVisible: true,
        title: line.label,
      }),
    );

    return () => {
      const currentSeries = candleSeriesRef.current;
      if (!currentSeries) return;
      for (const line of priceLinesRef.current) {
        currentSeries.removePriceLine(line);
      }
      priceLinesRef.current = [];
    };
  }, [rangeLines, interval, symbolKey]);

  return <div ref={containerRef} style={{ width: '100%', flex: 1, minHeight: 0 }} />;
}

import { authenticatedFetch } from '../../lib/apiClient';

export async function adminNotesRequest<T = unknown>(
  method: string,
  body?: object,
): Promise<T | null> {
  const response = await authenticatedFetch('/api/admin/notes', {
    method,
    headers: { 'Content-Type': 'application/json' },
    ...(body ? { body: JSON.stringify(body) } : {}),
  });

  if (!response) throw new Error('로그인이 필요합니다.');

  if (!response.ok) {
    const detail = await response.text();
    throw new Error(detail || `HTTP ${response.status}`);
  }
  if (response.status === 204) return null;
  return (await response.json()) as T;
}

import { useCallback, useEffect, useState } from 'react';
import { adminNotesRequest } from '../api';
import type { Note } from '../types';

interface UseAdminNotesResult {
  notes: Note[];
  loading: boolean;
  message: string | null;
  setMessage: (message: string | null) => void;
  loadNotes: () => Promise<void>;
}

async function fetchNotes(): Promise<Note[]> {
  const result = await adminNotesRequest<{ data?: Note[] }>('GET');
  return Array.isArray(result?.data) ? result.data : [];
}

function getLoadErrorMessage(error: unknown): string {
  return `목록 조회 실패: ${error instanceof Error ? error.message : String(error)}`;
}

export function useAdminNotes(): UseAdminNotesResult {
  const [notes, setNotes] = useState<Note[]>([]);
  const [loading, setLoading] = useState(true);
  const [message, setMessage] = useState<string | null>(null);

  const loadNotes = useCallback(async () => {
    setLoading(true);
    try {
      setNotes(await fetchNotes());
    } catch (error) {
      setMessage(getLoadErrorMessage(error));
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    let active = true;

    fetchNotes()
      .then((result) => {
        if (active) setNotes(result);
      })
      .catch((error: unknown) => {
        if (active) setMessage(getLoadErrorMessage(error));
      })
      .finally(() => {
        if (active) setLoading(false);
      });

    return () => {
      active = false;
    };
  }, []);

  return { notes, loading, message, setMessage, loadNotes };
}

import type { Session } from '@supabase/supabase-js';
import { supabase } from '../../../lib/supabase';

interface AuthSessionResult {
  session: Session | null;
  authLoading: boolean;
}

export function useAuthSession(): AuthSessionResult {
  const [session, setSession] = useState<Session | null>(null);
  const [authLoading, setAuthLoading] = useState(true);

  useEffect(() => {
    let active = true;

    supabase.auth.getSession().then(({ data }) => {
      if (!active) return;
      setSession(data.session);
      setAuthLoading(false);

    return () => {
      active = false;
      subscription.subscription.unsubscribe();
    };
  }, []);

  return { session, authLoading };

  response: string;
}

interface ChatActionDefinition {
  action_id: string;
  command: string;
  aliases: string[];
  example: string;

type ConnectionStatus = 'connecting' | 'connected' | 'reconnecting';

let cachedChatCommands: ChatCommandDefinition[] | null = null;
let chatCommandsRequest: Promise<ChatCommandDefinition[]> | null = null;
let cachedChatActions: ChatActionDefinition[] | null = null;
let chatActionsRequest: Promise<ChatActionDefinition[]> | null = null;

function getErrorMessage(error: unknown): string {
  return error instanceof Error ? error.message : '알 수 없는 오류가 발생했습니다.';
}

function isChatCommandDefinition(value: unknown): value is ChatCommandDefinition {
  if (typeof value !== 'object' || value === null) return false;
  const command = value as Record<string, unknown>;
  return typeof command.command === 'string' && typeof command.response === 'string';
}

function isChatCommandList(value: unknown): value is ChatCommandDefinition[] {
  return Array.isArray(value) && value.every(isChatCommandDefinition);
}

function loadChatCommands(): Promise<ChatCommandDefinition[]> {
  if (cachedChatCommands) return Promise.resolve(cachedChatCommands);
  if (chatCommandsRequest) return chatCommandsRequest;

  const request = (async () => {
    const response = await fetch('/api/chat/commands');
    if (!response.ok) {
      throw new Error(`명령어 목록 요청 실패 (HTTP ${response.status})`);
    }

    const commandData: unknown = await response.json();
    if (!isChatCommandList(commandData)) {
      throw new Error('채팅 서버가 올바르지 않은 명령어 목록을 반환했습니다.');
    }

    cachedChatCommands = commandData;
    return commandData;
  })();
  chatCommandsRequest = request;
  void request.then(
    () => {
      if (chatCommandsRequest === request) chatCommandsRequest = null;
    },
    () => {
      if (chatCommandsRequest === request) chatCommandsRequest = null;
    },
  );
  return request;
}

function isChatActionDefinition(value: unknown): value is ChatActionDefinition {
  if (typeof value !== 'object' || value === null) return false;
  const action = value as Record<string, unknown>;
  return (
    typeof action.action_id === 'string' &&
    typeof action.command === 'string' &&
    Array.isArray(action.aliases) &&
    action.aliases.every((alias) => typeof alias === 'string') &&
    typeof action.example === 'string'
  );
}

function isChatActionList(value: unknown): value is ChatActionDefinition[] {
  return Array.isArray(value) && value.every(isChatActionDefinition);
}

function loadChatActions(): Promise<ChatActionDefinition[]> {
  if (cachedChatActions) return Promise.resolve(cachedChatActions);
  if (chatActionsRequest) return chatActionsRequest;

  const request = (async () => {
    const response = await fetch('/api/chat/actions');
    if (!response.ok) {
      throw new Error(`행동 명령어 목록 요청 실패 (HTTP ${response.status})`);
    }
    const actionData: unknown = await response.json();
    if (!isChatActionList(actionData)) {
      throw new Error('채팅 서버가 올바르지 않은 행동 명령어 목록을 반환했습니다.');
    }
    cachedChatActions = actionData;
    return actionData;
  })();
  chatActionsRequest = request;
  void request.then(
    () => {
      if (chatActionsRequest === request) chatActionsRequest = null;
    },
    () => {
      if (chatActionsRequest === request) chatActionsRequest = null;
    },
  );
  return request;
}

function isChatTicket(value: unknown): value is ChatTicket {
  if (typeof value !== 'object' || value === null) return false;
  const ticket = value as Record<string, unknown>;
  return (
    typeof ticket.ticket === 'string' &&
    typeof ticket.nickname === 'string' &&
    typeof ticket.expires_in_seconds === 'number'
  );
}

function isChatMessage(value: unknown): value is ChatMessage {
  if (typeof value !== 'object' || value === null) return false;
  const message = value as Record<string, unknown>;
  return (
    message.type === 'message' &&
    typeof message.id === 'string' &&
    typeof message.nickname === 'string' &&
    typeof message.text === 'string' &&
    (message.message_kind === undefined ||
      message.message_kind === 'chat' ||
      message.message_kind === 'social') &&
    (message.action_id === undefined || typeof message.action_id === 'string') &&
    typeof message.sent_at === 'string'
  );
}

function getChatWebSocketUrl(ticket: string): string {
  const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
  const url = new URL(`${protocol}//${window.location.host}/ws/chat`);
  url.searchParams.set('ticket', ticket);
  return url.toString();
}

function formatTime(value: string): string {
  const date = new Date(value);
  return Number.isNaN(date.getTime()) ? value : date.toLocaleTimeString();
}

function usePublicChat(enabled: boolean) {
  const [, setStatus] = useState<ConnectionStatus>('connecting');
  const [nickname, setNickname] = useState('');
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [onlineCount, setOnlineCount] = useState(0);
  const [connectionError, setConnectionError] = useState('');
  const [, setConnectionClosed] = useState(false);
  const [sendError, setSendError] = useState('');
  const socketRef = useRef<WebSocket | null>(null);
  // StrictMode replays effects in development; share the pending POST to avoid burning an alias.
  const ticketRequestRef = useRef<Promise<ChatTicket> | null>(null);

  useEffect(() => {
    if (!enabled) {
      setStatus('connecting');
      return;
    }

    let disposed = false;
    let connectedBefore = false;
    let connecting = false;
    let reconnectDelay = INITIAL_RECONNECT_DELAY_MS;
    let reconnectTimer: number | null = null;
    let socket: WebSocket | null = null;

    const requestTicket = () => {
      if (ticketRequestRef.current) return ticketRequestRef.current;

      const request = (async () => {
        const response = await fetch('/api/chat/ticket', { method: 'POST' });
        if (!response.ok) {
          throw new Error(`티켓 발급 실패 (HTTP ${response.status})`);
        }

        const ticketData: unknown = await response.json();
        if (!isChatTicket(ticketData)) {
          throw new Error('채팅 서버가 올바르지 않은 연결 티켓을 반환했습니다.');
        }
        return ticketData;
      })();
      ticketRequestRef.current = request;
      void request.then(
        () => {
          if (ticketRequestRef.current === request) ticketRequestRef.current = null;
        },
        () => {
          if (ticketRequestRef.current === request) ticketRequestRef.current = null;
        },
      );
      return request;
    };

    const scheduleReconnect = () => {
      if (disposed || !enabled || reconnectTimer !== null) return;
      setStatus('reconnecting');
      reconnectTimer = window.setTimeout(() => {
        reconnectTimer = null;
        void connect();
      }, reconnectDelay);
      reconnectDelay = Math.min(reconnectDelay * 2, MAX_RECONNECT_DELAY_MS);
    };

    const connect = async () => {
      if (disposed || !enabled || connecting) return;
      connecting = true;
      if (connectedBefore) setStatus('reconnecting');

      try {
        const ticketData = await requestTicket();
        if (disposed) return;

        setNickname(ticketData.nickname);
        const nextSocket = new WebSocket(getChatWebSocketUrl(ticketData.ticket));
        socket = nextSocket;
        socketRef.current = nextSocket;

        nextSocket.onopen = () => {
          if (disposed) {
            nextSocket.close();
            return;
          }
          connectedBefore = true;
          reconnectDelay = INITIAL_RECONNECT_DELAY_MS;
          setConnectionClosed(false);
          setStatus('connected');
          setConnectionError('');
        };

        nextSocket.onmessage = (event: MessageEvent<unknown>) => {
          if (disposed) return;
          try {
            if (typeof event.data !== 'string') {
              throw new Error('채팅 서버에서 지원하지 않는 메시지 형식을 받았습니다.');
            }
            const data: unknown = JSON.parse(event.data);
            if (
              typeof data === 'object' &&
              data !== null &&
              'type' in data &&
              data.type === 'presence' &&
              'online_count' in data &&
              typeof data.online_count === 'number' &&
              Number.isInteger(data.online_count) &&
              data.online_count >= 0
            ) {
              setOnlineCount(data.online_count);
              return;
            }
            if (isChatMessage(data)) {
              setMessages((current) => [...current, data].slice(-MAX_MESSAGES));
              return;
            }
            if (
              typeof data === 'object' &&
              data !== null &&
              'type' in data &&
              data.type === 'error'
            ) {
              const code =
                'code' in data && typeof data.code === 'string' ? data.code : 'unknown';
              setConnectionError(`채팅 서버 오류: ${code}`);
              return;
            }
            throw new Error('채팅 서버에서 올바르지 않은 메시지를 받았습니다.');
          } catch (error) {
            setConnectionError(getErrorMessage(error));
          }
        };

        nextSocket.onerror = () => {
          if (disposed) return;
          setConnectionError('채팅 연결에 문제가 있습니다. 다시 연결합니다.');
          nextSocket.close();
        };

        nextSocket.onclose = () => {
          if (disposed || socketRef.current !== nextSocket) return;
          socketRef.current = null;
          socket = null;
          setConnectionClosed(true);
          setOnlineCount(0);
          scheduleReconnect();
        };
      } catch (error) {
        if (!disposed) {
          setConnectionError(getErrorMessage(error));
          scheduleReconnect();
        }
      } finally {
        connecting = false;
      }
    };

    void connect();
    return () => {
      disposed = true;
      if (reconnectTimer !== null) window.clearTimeout(reconnectTimer);
      socketRef.current = null;
      socket?.close();
    };
  }, [enabled]);

  const sendMessage = useCallback((text: string): boolean => {
    const trimmed = text.trim();
    if (!trimmed) return false;
    if (Array.from(trimmed).length > MAX_MESSAGE_CHARACTERS) {
      setSendError(`메시지는 ${MAX_MESSAGE_CHARACTERS}자까지 입력할 수 있습니다.`);
      return false;
    }

    const socket = socketRef.current;
    if (!socket || socket.readyState !== WebSocket.OPEN) {
      setSendError('채팅 서버와 연결되어 있지 않습니다.');
      return false;
    }

    try {
      socket.send(JSON.stringify({ type: 'send', text: trimmed }));
      setSendError('');
      return true;
    } catch (error) {
      setSendError(`메시지 전송 실패: ${getErrorMessage(error)}`);
      return false;
    }
  }, []);

  return {
    connectionError,
    messages,
    onlineCount,
    nickname,
    sendError,
    sendMessage,
    setSendError,
  };
}

export default function ChatPage({ active, backendReady }: { active: boolean; backendReady: boolean }) {
  const {
    connectionError,
    messages,
    onlineCount,
    nickname,
    sendError,
    sendMessage,
    setSendError,
  } = usePublicChat(backendReady);
  const [text, setText] = useState('');
  const [commands, setCommands] = useState<ChatCommandDefinition[]>([]);
  const [actions, setActions] = useState<ChatActionDefinition[]>([]);
  const [actionsOpen, setActionsOpen] = useState(false);
  const [actionLoadError, setActionLoadError] = useState('');
  const [commandLoadError, setCommandLoadError] = useState('');
  const [commandPickerDismissed, setCommandPickerDismissed] = useState(false);
  const [activeCommandIndex, setActiveCommandIndex] = useState(0);
  const messageListRef = useRef<HTMLOListElement>(null);
  const messageInputRef = useRef<HTMLInputElement>(null);
  const commandOptionRefs = useRef<Array<HTMLButtonElement | null>>([]);

  useEffect(() => {
    if (!active) return;
    const list = messageListRef.current;
    if (list) list.scrollTop = list.scrollHeight;
  }, [active, messages]);

  useEffect(() => {
    if (!backendReady) return;
    let disposed = false;
    void loadChatCommands().then(
      (loadedCommands) => {
        if (!disposed) setCommands(loadedCommands);
      },
      (error: unknown) => {
        if (!disposed) {
          setCommandLoadError(
            `명령어 목록을 불러오지 못했습니다: ${getErrorMessage(error)}`,
          );
        }
      },
    );
    return () => {
      disposed = true;
    };
  }, [backendReady]);

  useEffect(() => {
    if (!backendReady) return;
    let disposed = false;
    void loadChatActions().then(
      (loadedActions) => {
        if (!disposed) setActions(loadedActions);
      },
      (error: unknown) => {
        if (!disposed) {
          setActionLoadError(
            `행동 명령어 목록을 불러오지 못했습니다: ${getErrorMessage(error)}`,
          );
        }
      },
    );
    return () => {
      disposed = true;
    };
  }, [backendReady]);

  const commandSuggestions =
    !commandPickerDismissed && text.startsWith('/')
      ? commands.filter(({ command }) => command.startsWith(text))
      : [];
  const activeSuggestionIndex = Math.min(
    activeCommandIndex,
    Math.max(commandSuggestions.length - 1, 0),
  );

  useEffect(() => {
    if (!commandSuggestions.length) return;
    commandOptionRefs.current[activeSuggestionIndex]?.scrollIntoView({ block: 'nearest' });
  }, [activeSuggestionIndex, commandSuggestions.length]);

  const handleSubmit = (event: FormEvent<HTMLFormElement>) => {
    event.preventDefault();
    if (sendMessage(text)) setText('');
  };

  const selectCommand = (command: ChatCommandDefinition) => {
    const sent = sendMessage(command.command);
    setText(sent ? '' : command.command);
    setCommandPickerDismissed(true);
    setActiveCommandIndex(0);
    messageInputRef.current?.focus();
  };

  const handleComposerKeyDown = (event: KeyboardEvent<HTMLInputElement>) => {
    if (!commandSuggestions.length) {
      if (event.key === 'Escape' && text.startsWith('/')) {
        setCommandPickerDismissed(true);
      }
      return;
    }

    if (event.key === 'ArrowDown') {
      event.preventDefault();
      setActiveCommandIndex((index) => (index + 1) % commandSuggestions.length);
    } else if (event.key === 'ArrowUp') {
      event.preventDefault();
      setActiveCommandIndex(
        (index) => (index - 1 + commandSuggestions.length) % commandSuggestions.length,
      );
    } else if (event.key === 'Enter') {
      event.preventDefault();
      selectCommand(commandSuggestions[activeSuggestionIndex]);
    } else if (event.key === 'Escape') {
      event.preventDefault();
      setCommandPickerDismissed(true);
    }
  };

  return (
    <main className="chat-page">
      {connectionError && (
        <p className="chat-error" role="alert">
          {connectionError}
        </p>
      )}

      <p className="chat-online-count" aria-live="polite">현재 접속자 {onlineCount}명</p>

          <li className="chat-empty">새 메시지를 기다리는 중입니다.</li>
        ) : (
          messages.map((message, index) => {
            const isContinuation =
              index > 0 && messages[index - 1]?.nickname === message.nickname;

              >
                {!isContinuation && (
                  <div className="chat-message-meta">
                    <strong>{message.nickname}</strong>
                  </div>
                )}
                <div className="chat-message-row">
                  <p>

                    {message.message_kind === 'social' && message.action_id && (

            );
          })
        )}
      </ol>

      <form className="chat-composer" onSubmit={handleSubmit}>

          <div

                ref={(element) => {

                }}
                role="option"

        <label className="chat-visually-hidden" htmlFor="chat-message-input">

function getWebSocketUrl(): string {
  const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
  return `${protocol}//${window.location.host}/ws`;
}

function numeric(value: string): number {
  return Number.parseFloat(value.replace(/,/g, '').trim());
}

export function useMarketWebSocket({
  enabled,
  interval,
  selectedCode,
  setData,
  setCandles,
  onRangeUpdate,
  onRangeAlert,
}: UseMarketWebSocketOptions): boolean {
  const [isConnected, setIsConnected] = useState(false);
  const selectedCodeRef = useRef<string | null>(selectedCode);
  const intervalRef = useRef(interval);
  const alertedRangeKeysRef = useRef(new Set<string>());
  const rangeSnapshotRef = useRef<RangeMonitorSnapshot | null>(null);
  useLayoutEffect(() => {
    intervalRef.current = interval;
  }, [interval]);

  useEffect(() => {
    selectedCodeRef.current = selectedCode;
  }, [selectedCode]);

  useEffect(() => {
    if (!enabled) {
      setIsConnected(false);
      return;
    }

    let socket: WebSocket | null = null;
    let reconnectTimer: ReturnType<typeof window.setTimeout> | null = null;
    let reconnectDelay = RECONNECT_INITIAL_DELAY_MS;
    let disposed = false;
    const url = getWebSocketUrl();

    const connect = () => {
      if (disposed || !enabled) return;

      setIsConnected(false);
      const nextSocket = new WebSocket(url);
      socket = nextSocket;

      nextSocket.onopen = () => {
        if (disposed || socket !== nextSocket) {
          nextSocket.close();
          return;
        }
        reconnectDelay = RECONNECT_INITIAL_DELAY_MS;
        setIsConnected(true);
      };

      nextSocket.onmessage = (event) => {
        try {
          const message: unknown = JSON.parse(event.data);
          if (
            typeof message === 'object' &&
            message !== null &&
            'type' in message &&
            message.type === 'snapshot' &&
            'data' in message &&
            typeof message.data === 'object' &&
            message.data !== null &&
            'regular' in message.data &&
            'mini' in message.data
          ) {
            if ('range' in message && typeof message.range === 'object' && message.range !== null) {
              const range = message.range as RangeMonitorSnapshot;
              onRangeUpdate(range);
              rangeSnapshotRef.current = range;
              for (const session of [range.day, range.night]) {
                if (!session.date) continue;
                if (session.high_alerted) alertedRangeKeysRef.current.add(`${session.date}:${session.session}:high`);
                if (session.low_alerted) alertedRangeKeysRef.current.add(`${session.date}:${session.session}:low`);
              }
            }
            // 유효한 가격이 포함된 스냅샷만 적용합니다. 빈 스냅샷이면 REST 재시도를 계속합니다.
            const snapshot = message.data as PricesResponse;
            if (snapshot.regular?.ok || snapshot.mini?.ok) {
              setData(snapshot);
            }
            return;
          }

          if (
            typeof message !== 'object' ||
            message === null ||
            !('type' in message) ||
            message.type !== 'tick'
          ) return;
          const tick = message as RealtimeTick;
          if (tick.range) {
            const previousRange = rangeSnapshotRef.current;
            for (const session of [tick.range.day, tick.range.night]) {
              const previousSession = previousRange?.[session.session];
              for (const direction of ['high', 'low'] as const) {
                const isAlerted = direction === 'high' ? session.high_alerted : session.low_alerted;
                const wasAlerted = direction === 'high'
                  ? previousSession?.high_alerted
                  : previousSession?.low_alerted;
                if (!isAlerted || wasAlerted) continue;

                const alertKey = `${session.date ?? tick.date}:${session.session}:${direction}`;
                if (alertedRangeKeysRef.current.has(alertKey)) continue;
                alertedRangeKeysRef.current.add(alertKey);

                const matchingEvent = tick.range_alert
                  && tick.range_alert.session === session.session
                  && tick.range_alert.direction === direction
                  ? tick.range_alert
                  : null;
                const referencePrice = direction === 'high' ? session.high : session.low;
                const fallbackAlert: RangeAlert = {
                  session: session.session,
                  direction,
                  price: numeric(tick.price),
                  reference_price: referencePrice ?? numeric(tick.price),
                  message: direction === 'high'
                    ? '시세가 고가 돌파 했습니다. 숏 포지션은 위험 관리 하세요.'
                    : '시세가 저가를 깼습니다. 롱 포지션은 위험 관리 하세요.',
                };
                onRangeAlert(matchingEvent ?? fallbackAlert);
              }
            }
            rangeSnapshotRef.current = tick.range;
            onRangeUpdate(tick.range);
          } else if (tick.range_alert) {
            const alertKey = `${tick.date}:${tick.range_alert.session}:${tick.range_alert.direction}`;
            if (!alertedRangeKeysRef.current.has(alertKey)) {
              alertedRangeKeysRef.current.add(alertKey);
              onRangeAlert(tick.range_alert);
            }
          }

          setData((previous) => {
            if (!previous) return previous;

            const updateOne = (result: PricesResponse['regular']) => {
              if (tick.market_session === 'night') {
                const night = result.night;
                if (!night?.ok || !night.data || night.data.code !== tick.code) {
                  return result;
                }
                const current = numeric(tick.price);
                const high = numeric(night.data.high);
                const low = numeric(night.data.low);
                return {
                  ...result,
                  night: {
                    ...night,
                    data: {
                      ...night.data,
                      price: tick.price,
                      change: tick.change,
                      change_rate: tick.change_rate,
                      high: Number.isFinite(high) ? String(Math.max(high, current)) : tick.price,
                      low: Number.isFinite(low) ? String(Math.min(low, current)) : tick.price,
                      volume: tick.volume,
                    },
                  },
                };
              }

              if (!result.ok || !result.data || result.data.code !== tick.code) {
                return result;
              }
              return {
                ...result,
                data: {
                  ...result.data,
                  price: tick.price,
                  change: tick.change,
                  change_rate: tick.change_rate,
                  volume: tick.volume,
                  cached_at: Math.floor(Date.now() / 1000),
                },
              };
            };

            return {
              regular: updateOne(previous.regular),
              mini: updateOne(previous.mini),
            };
          });

          // 선택 종목의 주간/야간 체결을 모두 현재 차트에 반영합니다.
          if (tick.code !== selectedCodeRef.current) return;
          setCandles((previous) =>
            applyTickToCandles(previous, tick, intervalRef.current),
          );
        } catch {
          // Ignore malformed or unexpected WebSocket messages.
        }
      };

      nextSocket.onclose = () => {
        if (disposed || !enabled || socket !== nextSocket) return;

        socket = null;
        setIsConnected(false);
        reconnectTimer = window.setTimeout(connect, reconnectDelay);
        reconnectDelay = Math.min(reconnectDelay * 2, RECONNECT_MAX_DELAY_MS);
      };

      nextSocket.onerror = () => {
        if (disposed || socket !== nextSocket) return;

        setIsConnected(false);
        nextSocket.close();
      };
    };

    connect();

    return () => {
      disposed = true;
      if (reconnectTimer !== null) {
        window.clearTimeout(reconnectTimer);
      }
      socket?.close();
    };
  }, [enabled, setData, setCandles, onRangeUpdate, onRangeAlert]);

  return isConnected;
}

import type { PricesResponse } from '../../../types';
import type { CandleData, SymbolKind } from '../types';

interface UsePricesResult {

  setData: Dispatch<SetStateAction<PricesResponse | null>>;

  setCandles: Dispatch<SetStateAction<CandleData[]>>;

interface CandleState {
  key: string;
  data: CandleData[];
}

export function usePrices(interval: string, selected: SymbolKind, backendReady: boolean): UsePricesResult {
  const [data, setData] = useState<PricesResponse | null>(null);
  const [candleState, setCandleState] = useState<CandleState>({
    key: '',
    data: [],
  });

  useEffect(() => {
    if (!backendReady || data?.regular?.ok || data?.mini?.ok) return;
    const controller = new AbortController();

    const fetchPrices = async () => {
      let retryDelay = 1_000;

      while (!controller.signal.aborted) {
        try {
          const response = await fetch('/api/prices', {
            signal: controller.signal,
          });
          if (!response.ok) throw new Error(`HTTP ${response.status}`);

          const json: PricesResponse = await response.json();
          const hasQuote = Boolean(json.regular?.ok || json.mini?.ok);
          if (hasQuote && !controller.signal.aborted) {
            // WebSocket 스냅샷이 먼저 도착했다면 REST 응답이 실시간 값을 덮어쓰지 않습니다.
            setData((previous) => previous ?? json);
            return;
          }
        } catch (error) {
          if (controller.signal.aborted) return;
          console.error('prices fetch 실패, 재시도합니다:', error);
        }

        await new Promise((resolve) => window.setTimeout(resolve, retryDelay));
        retryDelay = Math.min(retryDelay * 2, 10_000);
      }
    };

    void fetchPrices();
    return () => controller.abort();
  }, [backendReady, data]);

  const selectedResult = selected === 'regular' ? data?.regular : data?.mini;
  const selectedCode =
    selectedResult?.ok && selectedResult.data ? selectedResult.data.code : null;
  const candleKey = selectedCode ? `${selectedCode}:${interval}` : '';

  // 현재 종목/주기와 일치하는 데이터만 차트에 전달해 이전 요청의 캔들을 숨긴다.
  const candles = candleState.key === candleKey && candleKey
    ? candleState.data
    : [];

  const setCandles = useCallback<Dispatch<SetStateAction<CandleData[]>>>(
    (action) => {
      setCandleState((previous) => {
        const current = previous.key === candleKey ? previous.data : [];
        const next = typeof action === 'function' ? action(current) : action;
        return { key: candleKey, data: next };
      });
    },
    [candleKey],
  );

  useEffect(() => {
    if (!backendReady || !selectedCode) return;

    const controller = new AbortController();
    const params = new URLSearchParams({
      code: selectedCode,
      interval,
      date: 'both',
    });

    const fetchCandles = async () => {
      try {
        const response = await fetch(`/api/candles?${params.toString()}`, {
          signal: controller.signal,
        });
        if (!response.ok) return;

        const json = await response.json();
        if (!json.ok || controller.signal.aborted) return;

        const raw: CandleData[] = json.data;
        setCandleState((previous) => {
          const current = previous.key === candleKey ? previous.data : [];
          const merged = new Map<string, CandleData>();
          for (const candle of raw) {
            merged.set(`${candle.date}:${candle.time}`, candle);
          }
          // REST 누적 거래량과 조회 이후 실시간 체결을 함께 보존합니다.
          for (const candle of current) {
            const key = `${candle.date}:${candle.time}`;
            const apiCandle = merged.get(key);
            merged.set(
              key,
              apiCandle
                ? {
                    ...apiCandle,
                    high: Math.max(apiCandle.high, candle.high),
                    low: Math.min(apiCandle.low, candle.low),
                    close: candle.close,
                    volume: Math.max(apiCandle.volume, candle.volume),
                  }
                : candle,
            );
          }
          return {
            key: candleKey,
            data: mapCandlesForChart([...merged.values()]),
          };
        });
      } catch (error) {
        if (!controller.signal.aborted) {
          console.error('candles fetch 실패:', error);
        }
      }
    };

    void fetchCandles();
    // 서버가 18시 이후 REST 야간 분봉을 비동기로 채우는 경우를 반영하고,
    // 조회 사이에 새로 생성된 분봉도 주기적으로 동기화합니다.
    const refreshTimer = window.setInterval(() => {
      void fetchCandles();
    }, 10_000);

    return () => {
      controller.abort();
      window.clearInterval(refreshTimer);
    };
  }, [backendReady, selectedCode, interval, candleKey]);

  return { data, setData, candles, setCandles, selectedCode };
}

import type { RealtimeTick } from '../../../types';
import type { CandleData } from '../types';

export function intervalToMinutes(interval: string): number {
  return parseInt(interval.replace('m', ''), 10);
}

export function mapCandlesForChart(raw: CandleData[]): CandleData[] {
  // CM 야간 분봉은 자정 이후 시간을 24시 이상으로 표현합니다.
  // 영업일 날짜 + 확장 시각을 그대로 사용해야 주간장 뒤에 연속해서 정렬됩니다.
  return [...raw].sort((a, b) =>
    a.date === b.date
      ? a.time.localeCompare(b.time)
      : a.date.localeCompare(b.date),
  );
}

export function applyTickToCandles(
  candles: CandleData[],
  tick: RealtimeTick,
  interval: string,
): CandleData[] {
  const price = parseFloat(tick.price);
  const quantity = parseInt(tick.last_qty, 10);

  if (
    Number.isNaN(price) ||
    price <= 0 ||
    tick.time.length < 4 ||
    !/^\d{8}$/.test(tick.date)
  ) {
    return candles;
  }

  const hhmm = tick.time.slice(0, 4);
  const minutes = intervalToMinutes(interval);
  const hour = Number(hhmm.slice(0, 2));
  const minute = Number(hhmm.slice(2, 4));
  const groupKey = Math.floor((hour * 60 + minute) / minutes);
  // CandleStore도 각 interval 그룹의 첫 체결 분을 봉 시각으로 사용합니다.
  const candleTime = `${hhmm}00`;

  const updated = [...candles];
  if (updated.length === 0) {
    return [{
      date: tick.date,
      time: candleTime,
      open: price,
      high: price,
      low: price,
      close: price,
      volume: quantity,
    }];
  }

  const last = updated[updated.length - 1];
  if (last.date !== tick.date) {
    updated.push({
      date: tick.date,
      time: candleTime,
      open: price,
      high: price,
      low: price,
      close: price,
      volume: quantity,
    });
    return updated;
  }

  const lastHour = Number(last.time.slice(0, 2));
  const lastMinute = Number(last.time.slice(2, 4));
  const lastGroupKey = Math.floor((lastHour * 60 + lastMinute) / minutes);

  if (groupKey === lastGroupKey) {
    updated[updated.length - 1] = {
      ...last,
      high: Math.max(last.high, price),
      low: Math.min(last.low, price),
      close: price,
      volume: last.volume + quantity,
    };
  } else if (groupKey > lastGroupKey) {
    updated.push({
      date: tick.date,
      time: candleTime,
      open: price,
      high: price,
      low: price,
      close: price,
      volume: quantity,
    });
  }

  return updated;
}

interface HealthResponse {

const INITIAL_RETRY_DELAY_MS = 1_000;

export function useBackendHealth(): BackendHealthStatus {
  const [status, setStatus] = useState<BackendHealthStatus>('checking');

  useEffect(() => {
    let disposed = false;
    let timer: number | null = null;
    let retryDelay = INITIAL_RETRY_DELAY_MS;
    let controller: AbortController | null = null;

    const checkHealth = async () => {
      controller = new AbortController();

      try {
        const response = await fetch('/api/health', {
          method: 'GET',
          cache: 'no-store',
          signal: controller.signal,
        });
        const payload: HealthResponse = await response.json();

        if (disposed) return;

        if (response.ok && (payload.ready === true || payload.status === 'ok')) {
          setStatus('ready');
          retryDelay = INITIAL_RETRY_DELAY_MS;
          timer = window.setTimeout(checkHealth, READY_POLL_INTERVAL_MS);
          return;
        }

        if (payload.status === 'failed') {
          setStatus('failed');
          retryDelay = MAX_RETRY_DELAY_MS;
        } else {
          setStatus('checking');
          retryDelay = INITIAL_RETRY_DELAY_MS;
        }
      } catch {
        if (disposed) return;
        setStatus('unavailable');
        retryDelay = Math.min(retryDelay * 2, MAX_RETRY_DELAY_MS);
      }

      if (!disposed) {
        timer = window.setTimeout(checkHealth, retryDelay);
      }
    };

    void checkHealth();

    return () => {
      disposed = true;
      if (timer !== null) window.clearTimeout(timer);
      controller?.abort();
    };
  }, []);

  return status;
}

export async function authenticatedFetch(
  input: RequestInfo | URL,
  init: RequestInit = {},
): Promise<Response | null> {
  const { data: sessionData, error: sessionError } = await supabase.auth.getSession();
  if (sessionError) throw sessionError;

  const token = sessionData.session?.access_token;
  if (!token) return null;

  const headers = new Headers(init.headers);
  headers.set('Authorization', `Bearer ${token}`);

  return fetch(input, { ...init, headers });
}

function SignedAmount({ value }: { value: number }) {
  const [integer, fraction] = Math.abs(value).toFixed(2).split('.');
  const sign = value > 0 ? '+' : value < 0 ? '-' : '';
  return (
    <span className="delta-amount">
      <span className="delta-amount-sign">{sign}</span>
      <span className="delta-amount-integer">{integer}</span>
      <span className="delta-amount-decimal">.</span>
      <span className="delta-amount-fraction">{fraction}</span>
    </span>
  );

// 기준가 대비 차이를 부호와 함께 계산합니다.
function diffFrom(value: string, base: string) {
  const v = toNumber(value);
  const b = toNumber(base);
  if (!Number.isFinite(v) || !Number.isFinite(b)) return null;
  const diff = v - b;
  return {
    sign: diff > 0 ? '▲' : diff < 0 ? '▼' : '-',
    value: diff,
  };
}

function extractMonth(name: string): string {
  const m = name.match(/20(\d{2})(\d{2})/);
  if (m) return `20${m[1]}년 ${m[2]}월`;
  return '';
}

function formatPrice(value?: string): string {
  return value && value.trim() ? value : '-';
}

function DiffValue({ value, base, showRate = false, useCandleColors = false }: {
  value: string;
  base: string;
  showRate?: boolean;
  useCandleColors?: boolean;
}) {
  const diff = diffFrom(value, base);
  if (!diff) return null;
  const directionClass = diff.sign === '▲' ? 'up' : diff.sign === '▼' ? 'down' : 'flat';
  const color = useCandleColors
    ? directionClass === 'up'
      ? candleColors.up
      : directionClass === 'down'
        ? candleColors.down
        : undefined
    : undefined;
  const baseNumber = toNumber(base);
  const rate = Number.isFinite(baseNumber) && baseNumber !== 0
    ? (diff.value / baseNumber) * 100
    : Number.NaN;
  const rateText = rate > 0 ? `+${rate.toFixed(2)}` : rate.toFixed(2);
  return (
    <span className={`diff ${directionClass}`} style={{ color }}>
      <span className="delta-direction">{diff.sign}</span>
      <SignedAmount value={diff.value} />
      {showRate && Number.isFinite(rate) && (
        <span className="delta-rate">({rateText}%)</span>
      )}
    </span>
  );
}
